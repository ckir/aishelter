1//! Request signing and verification.
2//!
3//! This module defines [`SignedRequest`], which wraps an HTTP request with an
4//! Ed25519 signature so that the server can authenticate the calling agent
5//! and verify that the request body has not been tampered with (§11).
6//!
7//! The signing scheme produces a canonical string of the form
8//! `agent_id\ntimestamp\nnonce\nmethod\npath\nbody_sha256` that is then
9//! signed with the agent's Ed25519 private key.  The server reconstructs
10//! the same string and verifies it against the agent's registered public key.
11
12use crate::keypair::AgentKeypair;
13use ed25519_dalek::{Signature, SignatureError, Signer, Verifier, VerifyingKey};
14use serde::{Deserialize, Serialize};
15use sha2::{Digest, Sha256};
16
17/// HTTP header names used for agent authentication.
18pub const X_AGENT_ID: &str = "x-agent-id";
19pub const X_AGENT_TIMESTAMP: &str = "x-agent-timestamp";
20pub const X_AGENT_NONCE: &str = "x-agent-nonce";
21pub const X_AGENT_SIGNATURE: &str = "x-agent-signature";
22
23/// A signed HTTP request from an agent.
24///
25/// Every field is required for verification.  The signature covers the
26/// agent identity, a timestamp, a nonce, the HTTP method and path, and
27/// the SHA-256 hash of the request body (§11).
28#[derive(Debug, Clone, Serialize, Deserialize)]
29pub struct SignedRequest {
30    /// The agent ID that signed this request.
31    pub agent_id: String,
32    /// RFC 3339 timestamp when the request was signed.
33    pub timestamp: String,
34    /// Unique nonce to prevent replay attacks (§10).
35    pub nonce: String,
36    /// HTTP method (e.g. `"POST"`, `"GET"`).
37    pub method: String,
38    /// Request path (e.g. `"/v1/tasks"`).
39    pub path: String,
40    /// Hex-encoded SHA-256 hash of the request body.
41    pub body_sha256: String,
42    /// Hex-encoded Ed25519 signature over the canonical message string.
43    pub signature: String,
44}
45
46impl SignedRequest {
47    /// Compute the SHA-256 hash of a request body, returned as hex.
48    ///
49    /// This is used to bind the request body to the signature so that
50    /// any modification of the payload invalidates the signature.
51    pub fn hash_body(body: &[u8]) -> String {
52        // Initialize the SHA-256 hasher.
53        let mut hasher = Sha256::new();
54        // Feed the body bytes into the hasher.
55        hasher.update(body);
56        // Finalize and return the hex digest.
57        format!("{:x}", hasher.finalize())
58    }
59
60    /// Construct a [`SignedRequest`] by signing the canonical message string.
61    ///
62    /// # Arguments
63    ///
64    /// * `keypair` — the agent's Ed25519 keypair.
65    /// * `agent_id` — the agent's unique identifier.
66    /// * `method` — the HTTP method of the request.
67    /// * `path` — the request path.
68    /// * `body` — the raw request body bytes.
69    /// * `nonce` — a unique nonce for replay protection.
70    ///
71    /// # Canonical message format
72    ///
73    /// The signed string is `agent_id\ntimestamp\nnonce\nmethod\npath\nbody_sha256`,
74    /// where `timestamp` is the current UTC time in RFC 3339 format.
75    pub fn sign(
76        keypair: &AgentKeypair,
77        agent_id: &str,
78        method: &str,
79        path: &str,
80        body: &[u8],
81        nonce: &str,
82    ) -> Self {
83        // Hash the request body.
84        let body_sha256 = Self::hash_body(body);
85        // Record the current timestamp.
86        let timestamp = chrono::Utc::now().to_rfc3339();
87
88        // Build the canonical message string to sign.
89        let message = format!(
90            "{}\n{}\n{}\n{}\n{}\n{}",
91            agent_id, timestamp, nonce, method, path, body_sha256
92        );
93
94        // Sign the message with the agent's private key.
95        let sig = keypair.signing_key().sign(message.as_bytes());
96
97        Self {
98            agent_id: agent_id.to_string(),
99            timestamp,
100            nonce: nonce.to_string(),
101            method: method.to_string(),
102            path: path.to_string(),
103            body_sha256,
104            signature: hex::encode(sig.to_bytes()),
105        }
106    }
107
108    /// Verify the signature against the agent's public key.
109    ///
110    /// This reconstructs the canonical message string from the stored fields
111    /// and verifies the Ed25519 signature.  The server calls this after
112    /// looking up the agent's public key from the registry (§11).
113    ///
114    /// # Errors
115    ///
116    /// Returns [`SignatureError`] if the signature is malformed, the hex
117    /// decoding fails, or the Ed25519 verification rejects.
118    pub fn verify(&self, public_key: &VerifyingKey) -> Result<(), SignatureError> {
119        // Reconstruct the canonical message string from stored fields.
120        let message = format!(
121            "{}\n{}\n{}\n{}\n{}\n{}",
122            self.agent_id, self.timestamp, self.nonce, self.method, self.path, self.body_sha256
123        );
124
125        // Decode the hex signature into raw bytes.
126        let sig_bytes = hex::decode(&self.signature).map_err(|_| SignatureError::new())?;
127        // Construct the Ed25519 Signature from the byte slice.
128        let sig = Signature::try_from(sig_bytes.as_slice()).map_err(|_| SignatureError::new())?;
129        // Verify the signature against the reconstructed message.
130        public_key.verify(message.as_bytes(), &sig)
131    }
132}
