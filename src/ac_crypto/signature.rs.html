1//! Request signing and verification.
2//!
3//! This module defines [`SignedRequest`], which wraps an HTTP request with an
4//! Ed25519 signature so that the server can authenticate the calling agent
5//! and verify that the request body has not been tampered with (§11).
6//!
7//! The signing scheme produces a canonical string of the form
8//! `agent_id\ntimestamp\nnonce\nmethod\npath\nbody_sha256` that is then
9//! signed with the agent's Ed25519 private key.  The server reconstructs
10//! the same string and verifies it against the agent's registered public key.
11
12use crate::keypair::AgentKeypair;
13use ed25519_dalek::{Signature, SignatureError, Signer, Verifier, VerifyingKey};
14use serde::{Deserialize, Serialize};
15use sha2::{Digest, Sha256};
16
17/// HTTP header names used for agent authentication.
18pub const X_AGENT_ID: &str = "x-agent-id";
19pub const X_AGENT_TIMESTAMP: &str = "x-agent-timestamp";
20pub const X_AGENT_NONCE: &str = "x-agent-nonce";
21pub const X_AGENT_SIGNATURE: &str = "x-agent-signature";
22
23/// A signed HTTP request from an agent.
24///
25/// Every field is required for verification.  The signature covers the
26/// agent identity, a timestamp, a nonce, the HTTP method and path, and
27/// the SHA-256 hash of the request body (§11).
28#[derive(Debug, Clone, Serialize, Deserialize)]
29pub struct SignedRequest {
30    /// The agent ID that signed this request.
31    pub agent_id: String,
32    /// RFC 3339 timestamp when the request was signed.
33    pub timestamp: String,
34    /// Unique nonce to prevent replay attacks (§10).
35    pub nonce: String,
36    /// HTTP method (e.g. `"POST"`, `"GET"`).
37    pub method: String,
38    /// Request path (e.g. `"/v1/tasks"`).
39    pub path: String,
40    /// Hex-encoded SHA-256 hash of the request body.
41    pub body_sha256: String,
42    /// Hex-encoded Ed25519 signature over the canonical message string.
43    pub signature: String,
44}
45
46impl SignedRequest {
47    /// Compute the SHA-256 hash of a request body, returned as hex.
48    ///
49    /// This is used to bind the request body to the signature so that
50    /// any modification of the payload invalidates the signature.
51    pub fn hash_body(body: &[u8]) -> String {
52        // Initialize the SHA-256 hasher.
53        let mut hasher = Sha256::new();
54        // Feed the body bytes into the hasher.
55        hasher.update(body);
56        // Finalize and return the hex digest.
57        format!("{:x}", hasher.finalize())
58    }
59
60    /// Construct a [`SignedRequest`] by signing the canonical message string.
61    ///
62    /// # Arguments
63    ///
64    /// * `keypair` — the agent's Ed25519 keypair.
65    /// * `agent_id` — the agent's unique identifier.
66    /// * `method` — the HTTP method of the request.
67    /// * `path` — the request path.
68    /// * `body` — the raw request body bytes.
69    /// * `nonce` — a unique nonce for replay protection.
70    ///
71    /// # Canonical message format
72    ///
73    /// The signed string is `agent_id\ntimestamp\nnonce\nmethod\npath\nbody_sha256`,
74    /// where `timestamp` is the current UTC time in RFC 3339 format.
75    pub fn sign(
76        keypair: &AgentKeypair,
77        agent_id: &str,
78        method: &str,
79        path: &str,
80        body: &[u8],
81        nonce: &str,
82    ) -> Self {
83        Self::sign_at(
84            keypair,
85            agent_id,
86            method,
87            path,
88            body,
89            nonce,
90            &Self::format_timestamp(chrono::Utc::now()),
91        )
92    }
93
94    /// Like [`sign`](Self::sign) but with an explicit timestamp string.
95    ///
96    /// The signature covers `timestamp` exactly as given, byte for byte.
97    /// Production callers use [`sign`](Self::sign); this exists so tests can
98    /// produce stale or future-dated requests deterministically.
99    pub fn sign_at(
100        keypair: &AgentKeypair,
101        agent_id: &str,
102        method: &str,
103        path: &str,
104        body: &[u8],
105        nonce: &str,
106        timestamp: &str,
107    ) -> Self {
108        let mut req = Self {
109            agent_id: agent_id.to_string(),
110            timestamp: timestamp.to_string(),
111            nonce: nonce.to_string(),
112            method: method.to_string(),
113            path: path.to_string(),
114            body_sha256: Self::hash_body(body),
115            signature: String::new(),
116        };
117        let sig = keypair.signing_key().sign(req.canonical_message().as_bytes());
118        req.signature = hex::encode(sig.to_bytes());
119        req
120    }
121
122    /// The canonical timestamp form clients should emit: RFC 3339, UTC, `Z`
123    /// suffix, whole seconds (for example `2026-11-22T20:30:00Z`).
124    pub fn format_timestamp(at: chrono::DateTime<chrono::Utc>) -> String {
125        at.to_rfc3339_opts(chrono::SecondsFormat::Secs, true)
126    }
127
128    /// Parse a client-supplied timestamp for the clock-skew check.
129    ///
130    /// Accepts any valid RFC 3339 string (any offset, optional fractional
131    /// seconds) and rejects everything else, including bare Unix seconds.
132    /// The signature is always computed over the string as received, never
133    /// over a re-formatted value, so this is only used to read the instant.
134    pub fn parse_timestamp(
135        value: &str,
136    ) -> Result<chrono::DateTime<chrono::Utc>, chrono::ParseError> {
137        Ok(chrono::DateTime::parse_from_rfc3339(value)?.with_timezone(&chrono::Utc))
138    }
139
140    /// The exact byte string that is signed and verified.
141    ///
142    /// `path` is the request target including any query string, so a
143    /// signature also binds the query parameters.
144    fn canonical_message(&self) -> String {
145        format!(
146            "{}\n{}\n{}\n{}\n{}\n{}",
147            self.agent_id, self.timestamp, self.nonce, self.method, self.path, self.body_sha256
148        )
149    }
150
151    /// Verify the signature against the agent's public key.
152    ///
153    /// This reconstructs the canonical message string from the stored fields
154    /// and verifies the Ed25519 signature.  The server calls this after
155    /// looking up the agent's public key from the registry (§11).
156    ///
157    /// # Errors
158    ///
159    /// Returns [`SignatureError`] if the signature is malformed, the hex
160    /// decoding fails, or the Ed25519 verification rejects.
161    pub fn verify(&self, public_key: &VerifyingKey) -> Result<(), SignatureError> {
162        // Reconstruct the canonical message string from stored fields.
163        let message = self.canonical_message();
164
165        // Decode the hex signature into raw bytes.
166        let sig_bytes = hex::decode(&self.signature).map_err(|_| SignatureError::new())?;
167        // Construct the Ed25519 Signature from the byte slice.
168        let sig = Signature::try_from(sig_bytes.as_slice()).map_err(|_| SignatureError::new())?;
169        // Verify the signature against the reconstructed message.
170        public_key.verify(message.as_bytes(), &sig)
171    }
172}

174#[cfg(test)]
175mod tests {
176    use super::*;
177    use crate::keypair::{AgentKeypair, public_key_from_hex};
178
179    const AGENT: &str = "agent_test";
180
181    fn signed(path: &str, body: &[u8]) -> (AgentKeypair, SignedRequest) {
182        let kp = AgentKeypair::generate();
183        let req = SignedRequest::sign(&kp, AGENT, "POST", path, body, "nonce-1");
184        (kp, req)
185    }
186
187    #[test]
188    fn sign_emits_utc_z_whole_second_timestamp() {
189        let (_kp, req) = signed("/v1/messages", b"{}");
190        // e.g. 2026-11-22T20:30:00Z : 20 chars, ends in Z, no fraction, no offset.
191        assert_eq!(req.timestamp.len(), 20, "unexpected timestamp: {}", req.timestamp);
192        assert!(req.timestamp.ends_with('Z'), "unexpected timestamp: {}", req.timestamp);
193        assert!(!req.timestamp.contains('.'), "unexpected timestamp: {}", req.timestamp);
194        assert!(SignedRequest::parse_timestamp(&req.timestamp).is_ok());
195    }
196
197    #[test]
198    fn sign_then_verify_roundtrips() {
199        let (kp, req) = signed("/v1/messages", b"{\"a\":1}");
200        let pk = public_key_from_hex(&kp.public_key_hex()).unwrap();
201        assert!(req.verify(&pk).is_ok());
202    }
203
204    #[test]
205    fn sign_at_uses_the_given_timestamp_verbatim() {
206        let kp = AgentKeypair::generate();
207        let ts = "2026-01-02T03:04:05Z";
208        let req = SignedRequest::sign_at(&kp, AGENT, "GET", "/v1/x", b"", "n", ts);
209        assert_eq!(req.timestamp, ts);
210        let pk = public_key_from_hex(&kp.public_key_hex()).unwrap();
211        assert!(req.verify(&pk).is_ok());
212    }
213
214    #[test]
215    fn verify_rejects_a_changed_query_string() {
216        let kp = AgentKeypair::generate();
217        let mut req = SignedRequest::sign(&kp, AGENT, "GET", "/v1/messages?agent_id=a", b"", "n");
218        req.path = "/v1/messages?agent_id=b".to_string();
219        let pk = public_key_from_hex(&kp.public_key_hex()).unwrap();
220        assert!(req.verify(&pk).is_err());
221    }
222
223    #[test]
224    fn verify_rejects_a_changed_body_hash() {
225        let (kp, mut req) = signed("/v1/messages", b"original");
226        req.body_sha256 = SignedRequest::hash_body(b"tampered");
227        let pk = public_key_from_hex(&kp.public_key_hex()).unwrap();
228        assert!(req.verify(&pk).is_err());
229    }
230
231    #[test]
232    fn verify_rejects_a_changed_method() {
233        let (kp, mut req) = signed("/v1/messages", b"x");
234        req.method = "PUT".to_string();
235        let pk = public_key_from_hex(&kp.public_key_hex()).unwrap();
236        assert!(req.verify(&pk).is_err());
237    }
238
239    #[test]
240    fn verify_rejects_another_agents_key() {
241        let (_kp, req) = signed("/v1/messages", b"x");
242        let other = AgentKeypair::generate();
243        let pk = public_key_from_hex(&other.public_key_hex()).unwrap();
244        assert!(req.verify(&pk).is_err());
245    }
246
247    #[test]
248    fn parse_timestamp_accepts_z_offset_and_fraction() {
249        let z = SignedRequest::parse_timestamp("2026-09-05T14:00:00Z").unwrap();
250        let off = SignedRequest::parse_timestamp("2026-09-05T16:00:00+02:00").unwrap();
251        let frac = SignedRequest::parse_timestamp("2026-09-05T14:00:00.250Z").unwrap();
252        assert_eq!(z, off);
253        assert_eq!(z.timestamp(), frac.timestamp());
254    }

256    #[test]
257    fn parse_timestamp_rejects_unix_seconds_and_garbage() {
258        for bad in ["1795375064", "", "yesterday", "2026-09-05", "2026-09-05 14:00:00"] {
259            assert!(SignedRequest::parse_timestamp(bad).is_err(), "accepted {bad:?}");
260        }
