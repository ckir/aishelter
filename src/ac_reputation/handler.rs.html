13/// Application state for reputation routes.
14#[derive(Clone)]
15pub struct ReputationState {
16    pub pool: SharedPool,
17}

19/// Reputation scores response.
20#[derive(Serialize, utoipa::ToSchema)]
21pub struct ReputationResponse {
22    pub agent_id: String,
23    pub reliability: f64,
24    pub reliability_confidence: f64,
25    pub task_success: f64,
26    pub task_success_confidence: f64,
27    pub verification_accuracy: f64,
28    pub verification_accuracy_confidence: f64,
29    pub responsiveness: f64,
30    pub responsiveness_confidence: f64,
31    pub vwu_total: u64,
32}

34/// Contribution stats response.
35#[derive(Serialize, utoipa::ToSchema)]
36pub struct ContributionsResponse {
37    pub vwu_total: i64,
38    pub verified_tasks: i64,
39    pub validation_tasks: i64,
40    pub validation_accuracy: f64,
41}

43/// GET /{id}/reputation — Get reputation scores for an agent.
44#[utoipa::path(
45    get,
46    path = "/v1/agents/{id}/reputation",
47    tag = "reputation",
48    params(
49        ("id" = String, Path, description = "Agent ID"),
50    ),
51    responses(
52        (status = 200, description = "Reputation scores", body = ReputationResponse),
53        (status = 500, description = "Internal error"),
54    ),
55)]
56pub async fn get_reputation(
57    Path(agent_id): Path<String>,
58    State(state): State<ReputationState>,
59) -> Result<Json<ReputationResponse>, (StatusCode, String)> {
60    let pool = state.pool.load();
61    let service = ReputationService::new(pool);
62    let snapshot = service.get_reputation(&agent_id).await.map_err(|e| match e {
63        AcError::AgentNotFound(_) => (StatusCode::NOT_FOUND, e.to_string()),
64        AcError::Database(msg) => (StatusCode::INTERNAL_SERVER_ERROR, msg),
65        other => (StatusCode::INTERNAL_SERVER_ERROR, other.to_string()),
66    })?;
67
68    Ok(Json(ReputationResponse {
69        agent_id: snapshot.agent_id.0,
70        reliability: snapshot.reliability,
71        reliability_confidence: snapshot.reliability_confidence,
72        task_success: snapshot.task_success,
73        task_success_confidence: snapshot.task_success_confidence,
74        verification_accuracy: snapshot.verification_accuracy,
75        verification_accuracy_confidence: snapshot.verification_accuracy_confidence,
76        responsiveness: snapshot.responsiveness,
77        responsiveness_confidence: snapshot.responsiveness_confidence,
78        vwu_total: snapshot.vwu_total,
79    }))
80}

82/// GET /{id}/contributions — Get contribution stats for an agent.
83#[utoipa::path(
84    get,
85    path = "/v1/agents/{id}/contributions",
86    tag = "reputation",
87    params(
88        ("id" = String, Path, description = "Agent ID"),
89    ),
90    responses(
91        (status = 200, description = "Contribution stats", body = ContributionsResponse),
92        (status = 500, description = "Internal error"),
93    ),
94)]
95pub async fn get_contributions(
96    Path(agent_id): Path<String>,
97    State(state): State<ReputationState>,
98) -> Result<Json<ContributionsResponse>, (StatusCode, String)> {
99    let pool = state.pool.load();
100    let service = ReputationService::new(pool);
101    let stats = service
102        .get_contributions(&agent_id)
103        .await
104        .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, e.to_string()))?;
105
106    Ok(Json(ContributionsResponse {
107        vwu_total: stats.vwu_total,
108        verified_tasks: stats.verified_tasks,
109        validation_tasks: stats.validation_tasks,
110        validation_accuracy: stats.validation_accuracy,
111    }))
112}

114/// Build the reputation router.
115pub fn routes(pool: SharedPool) -> Router {
116    let state = ReputationState { pool };
117    Router::new()
118        .route("/{id}/reputation", get(get_reputation))
119        .route("/{id}/contributions", get(get_contributions))
120        .with_state(state)
121}
