1//! Axum HTTP server and route registration.
2//!
3//! This module assembles the top-level [`axum::Router`] that serves all
4//! Agent Commons API endpoints.  Each feature crate (registry, discovery,
5//! mailbox, tasks, validation, reputation) contributes its own nested
6//! sub-router via `pub fn routes(pool) -> Router`.
7//!
8//! The server also hosts the Scalar interactive API documentation at
9//! `/docs` and the raw OpenAPI specification at `/api/openapi.json`.
10
11use crate::middleware::rate_limit::{RateLimiter, rate_limit as rate_limit_mw};
12use ac_db::pool::SharedPool;
13use ac_metrics::middleware::record_metrics;
14use ac_metrics::readiness;
15use ac_metrics::registry::Metrics;
16use axum::{Router, extract::Extension, middleware::from_fn, routing::get};
17use http::header::HeaderName;
18use tower_http::request_id::{MakeRequestUuid, SetRequestIdLayer};
19use tower_http::trace::{DefaultMakeSpan, TraceLayer};
20use tracing::Level;
21use utoipa::OpenApi;
22use utoipa_scalar::{Scalar, Servable};
23
24use crate::routes::ApiDoc;

26/// Create the main application router with all sub-routes mounted.
27///
28/// The returned router includes:
29/// - System endpoints (`/v1/health`, `/v1/version`, `/metrics`)
30/// - Feature sub-routes under `/v1/{feature}`
31/// - HTTP request tracing via `tower-http`
32/// - Metrics middleware recording request counts and durations
33/// - Rate limiting with global and per-agent token buckets
34/// - OpenAPI JSON at `/api/openapi.json`
35/// - Interactive Scalar UI at `/docs`
36pub fn create_app(pool: SharedPool, metrics: Metrics, rate_limiter: RateLimiter) -> Router {
37    let app = Router::new()
38        .route("/v1/health", get(health_check))
39        .route("/v1/version", get(version_check))
40        .route("/metrics", get(metrics_handler))
41        .route(
42            "/v1/ready",
43            get({
44                let pool = pool.clone();
45                move || async move {
46                    let pg_pool = pool.load();
47                    if readiness::check(&pg_pool).await {
48                        (axum::http::StatusCode::OK, "ready")
49                    } else {
50                        (axum::http::StatusCode::SERVICE_UNAVAILABLE, "not ready")
51                    }
52                }
53            }),
54        )
55        .nest("/v1/agents", ac_registry::handler::routes(pool.clone()))
56        .nest("/", ac_directory::well_known::routes(pool.clone()))
57        .nest("/", ac_directory::handler::routes(pool.clone()))
58        .nest("/v1/discovery", ac_discovery::handler::routes(pool.clone()))
59        .nest("/v1/messages", ac_mailbox::handler::routes(pool.clone()))
60        .nest("/v1/tasks", ac_tasks::handler::routes(pool.clone()))
61        .nest("/v1/validations", ac_validation::handler::routes(pool.clone()))
62        .nest("/v1/agents", ac_reputation::handler::routes(pool.clone()))
63        .layer(from_fn(rate_limit_mw))
64        .layer(Extension(rate_limiter))
65        .layer(from_fn(record_metrics))
66        .layer(Extension(metrics.clone()))
67        .layer(
68            TraceLayer::new_for_http()
69                .make_span_with(DefaultMakeSpan::new().include_headers(true).level(Level::INFO)),
70        )
71        .layer(SetRequestIdLayer::new(HeaderName::from_static("x-request-id"), MakeRequestUuid));
72
73    // Merge OpenAPI spec and Scalar docs
74    let doc = ApiDoc::openapi();
75    app.route("/api/openapi.json", get(crate::routes::openapi))
76        .merge(Scalar::with_url("/docs", doc))
77}

79/// GET /v1/health — liveness probe.
80///
81/// Returns `"ok"` with HTTP 200 if the server is running.
82/// Used by container orchestrators and load balancers.
83#[utoipa::path(
84    get,
85    path = "/v1/health",
86    tag = "system",
87    responses(
88        (status = 200, description = "Service is healthy", body = String),
89    ),
90)]
91async fn health_check() -> &'static str {
92    "ok"
93}
94
95/// GET /v1/version — version and protocol info.
96///
97/// Returns the server version and the ACP protocol version it implements.
98/// Clients can use this to verify compatibility before making requests.
99#[utoipa::path(
100    get,
101    path = "/v1/version",
102    tag = "system",
103    responses(
104        (status = 200, description = "Version info", body = serde_json::Value,
105         example = json!({"version": "0.1.0", "protocol": "acp/1"})),
106    ),
107)]
108async fn version_check() -> axum::Json<serde_json::Value> {
109    axum::Json(serde_json::json!({
110        "version": "0.1.0",
111        "protocol": "acp/1"
112    }))
113}

115/// GET /metrics — Prometheus metrics endpoint.
116///
117/// Returns all registered metrics in Prometheus text exposition format.
118/// Consumed by Prometheus scrapers for observability dashboards.
119async fn metrics_handler(
120    Extension(metrics): Extension<Metrics>,
121) -> (axum::http::StatusCode, [(http::HeaderName, &'static str); 1], String) {
122    (
123        axum::http::StatusCode::OK,
124        [(http::header::CONTENT_TYPE, "text/plain; version=0.0.4; charset=utf-8")],
125        metrics.encode(),
126    )
127}
