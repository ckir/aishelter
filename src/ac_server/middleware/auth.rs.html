1//! Request authentication middleware for protected routes.
2//!
3//! Verifies the Ed25519 request signature (§10-11) and binds the verified
4//! agent id to the request as an [`AuthenticatedAgent`] extension. Handlers
5//! must take the caller's identity from that extension, never from the body,
6//! the query string or the `X-Agent-Id` header.
7//!
8//! Checks, in this order (cheap and side-effect free first):
9//!
10//! 1. credentials present (`X-Agent-Id`, `-Timestamp`, `-Nonce`, `-Signature`)
11//! 2. timestamp is RFC 3339 and within ±[`MAX_TIMESTAMP_SKEW_SECS`]
12//! 3. body is at most [`MAX_BODY_BYTES`] and is buffered for hashing
13//! 4. agent exists and is `REGISTERED` or `ACTIVE`
14//! 5. signature verifies over
15//!    `agent_id \n timestamp \n nonce \n method \n path_and_query \n sha256(body)`
16//! 6. nonce is consumed (single use), **only after** the signature verified,
17//!    so an unauthenticated caller cannot burn another agent's nonces
18//!
19//! Failure mapping: 401 missing credentials or failed authentication, 400
20//! malformed, stale or replayed credentials, 413 oversized body, 503 when
21//! the database is unavailable (fail closed).
22

39/// Maximum accepted difference between the client timestamp and server time.
40pub const MAX_TIMESTAMP_SKEW_SECS: i64 = 300;
41
42/// How long a consumed nonce is remembered.
43///
44/// A request dated up to `+MAX_TIMESTAMP_SKEW_SECS` ahead is still inside the
45/// skew window until `2 * MAX_TIMESTAMP_SKEW_SECS` after it was first seen, so
46/// the replay cache must outlive that or the request can be replayed.
47pub const NONCE_TTL_SECS: i64 = 2 * MAX_TIMESTAMP_SKEW_SECS;
48
49/// Largest request body the authentication layer will buffer.
50pub const MAX_BODY_BYTES: usize = 1024 * 1024;
51
52const MAX_ID_LEN: usize = 256;
53const MAX_NONCE_LEN: usize = 128;
54const MAX_SIGNATURE_LEN: usize = 256;
55
56/// Agent lifecycle states allowed to make signed requests.
57const ACTIVE_STATUSES: [&str; 2] = ["REGISTERED", "ACTIVE"];
58
59const TIMESTAMP_HELP: &str =
60    "X-Agent-Timestamp must be an RFC 3339 UTC timestamp, e.g. 2026-11-22T20:30:00Z";
61
62pub use ac_types::auth::AuthenticatedAgent;

64/// Why a request failed authentication.
65#[derive(Debug)]
66pub enum AuthError {
67    /// No `X-Agent-Id` header: the caller presented no credentials at all.
68    MissingCredentials,
69    /// Credentials present but malformed, stale, future-dated or replayed.
70    Malformed(String),
71    /// Unknown or inactive agent, or a signature that does not verify.
72    /// Deliberately one message, so the response is not an agent-existence oracle.
73    Rejected,
74    /// Body larger than [`MAX_BODY_BYTES`] (or unreadable).
75    PayloadTooLarge,
76    /// The database could not be reached; fail closed.
77    Unavailable,
78}
79
80impl IntoResponse for AuthError {
81    fn into_response(self) -> Response {
82        let (status, message) = match self {
83            AuthError::MissingCredentials => (
84                StatusCode::UNAUTHORIZED,
85                "authentication required: send X-Agent-Id, X-Agent-Timestamp, \
86                 X-Agent-Nonce and X-Agent-Signature"
87                    .to_string(),
88            ),
89            AuthError::Malformed(msg) => (StatusCode::BAD_REQUEST, msg),
90            AuthError::Rejected => (StatusCode::UNAUTHORIZED, "authentication failed".to_string()),
91            AuthError::PayloadTooLarge => (
92                StatusCode::PAYLOAD_TOO_LARGE,
93                format!("request body exceeds {MAX_BODY_BYTES} bytes"),
94            ),
95            AuthError::Unavailable => (
96                StatusCode::SERVICE_UNAVAILABLE,
97                "authentication temporarily unavailable".to_string(),
98            ),
99        };
100        let mut resp = (status, Json(serde_json::json!({ "error": message }))).into_response();
101        if status == StatusCode::UNAUTHORIZED {
102            resp.headers_mut()
103                .insert(header::WWW_AUTHENTICATE, HeaderValue::from_static("ACP1-Ed25519"));
104        }
105        resp
106    }
107}

109/// Read an optional header as UTF-8 text.
110fn header_text<'a>(headers: &'a HeaderMap, name: &str) -> Result<Option<&'a str>, AuthError> {
111    match headers.get(name) {
112        None => Ok(None),
113        Some(v) => v
114            .to_str()
115            .map(Some)
116            .map_err(|_| AuthError::Malformed(format!("{name} header is not valid text"))),
117    }
118}
119
120/// Read a required header, enforcing a maximum length.
121fn required_header<'a>(
122    headers: &'a HeaderMap,
123    name: &str,
124    max_len: usize,
125) -> Result<&'a str, AuthError> {
126    let value = header_text(headers, name)?
127        .ok_or_else(|| AuthError::Malformed(format!("missing {name} header")))?;
128    if value.is_empty() || value.len() > max_len {
129        return Err(AuthError::Malformed(format!("{name} header has an invalid length")));
130    }
131    Ok(value)
132}

134/// Authenticate the request and attach an [`AuthenticatedAgent`].
135///
136/// Layer it on protected routers with
137/// `from_fn_with_state(pool, authenticate)`.
138pub async fn authenticate(
139    State(pool): State<SharedPool>,
140    req: Request,
141    next: Next,
142) -> Result<Response, AuthError> {
143    let headers = req.headers();
144
145    // 1. Credentials. No agent id at all means "unauthenticated" (401);
146    //    an id with incomplete or garbled companions is a malformed request (400).
147    if header_text(headers, X_AGENT_ID)?.is_none() {
148        return Err(AuthError::MissingCredentials);
149    }
150    let agent_id = required_header(headers, X_AGENT_ID, MAX_ID_LEN)?.to_string();
151    let timestamp = required_header(headers, X_AGENT_TIMESTAMP, 64)?.to_string();
152    let nonce = required_header(headers, X_AGENT_NONCE, MAX_NONCE_LEN)?.to_string();
153    let signature = required_header(headers, X_AGENT_SIGNATURE, MAX_SIGNATURE_LEN)?.to_string();
154
155    // 2. Timestamp format and clock skew.
156    let parsed = SignedRequest::parse_timestamp(&timestamp)
157        .map_err(|_| AuthError::Malformed(TIMESTAMP_HELP.to_string()))?;
158    let skew = (chrono::Utc::now() - parsed).num_seconds().abs();
159    if skew > MAX_TIMESTAMP_SKEW_SECS {
160        return Err(AuthError::Malformed(format!(
161            "X-Agent-Timestamp is outside the allowed clock skew of {MAX_TIMESTAMP_SKEW_SECS}s"
162        )));
163    }
164
165    // The signed request target is what the client sent, query string
166    // included. A nested router has already stripped its prefix from
167    // `req.uri()`, so prefer the original URI that `nest` records.
168    let target = req
169        .extensions()
170        .get::<OriginalUri>()
171        .map(|original| original.0.clone())
172        .unwrap_or_else(|| req.uri().clone());
173    let path_and_query =
174        target.path_and_query().map(|pq| pq.as_str().to_owned()).unwrap_or_else(|| "/".to_owned());
175
176    // 3. Buffer the body (bounded) so it can be hashed, then restore it.
177    let (mut parts, body) = req.into_parts();
178    let body_bytes =
179        to_bytes(body, MAX_BODY_BYTES).await.map_err(|_| AuthError::PayloadTooLarge)?;
180
181    // 4. Agent lookup. A database failure fails closed.
182    let pg = pool.load();
183    let row: Option<(String, String)> =
184        sqlx::query_as("SELECT public_key, status FROM agents WHERE agent_id = $1")
185            .bind(&agent_id)
186            .fetch_optional(&pg)
187            .await
188            .map_err(|e| {
189                error!(error = %e, "auth: agent lookup failed");
190                AuthError::Unavailable
191            })?;
192    let Some((public_key_hex, status)) = row else {
193        warn!(agent_id = %agent_id, "auth: unknown agent");
194        return Err(AuthError::Rejected);
195    };
196    if !ACTIVE_STATUSES.contains(&status.as_str()) {
197        warn!(agent_id = %agent_id, status = %status, "auth: agent not active");
198        return Err(AuthError::Rejected);
199    }
200    let Ok(public_key) = public_key_from_hex(&public_key_hex) else {
201        warn!(agent_id = %agent_id, "auth: stored public key is not a valid Ed25519 key");
202        return Err(AuthError::Rejected);
203    };
204
205    // 5. Signature over the exact bytes the client sent.
206    let signed = SignedRequest {
207        agent_id: agent_id.clone(),
208        timestamp,
209        nonce: nonce.clone(),
210        method: parts.method.as_str().to_string(),
211        path: path_and_query,
212        body_sha256: SignedRequest::hash_body(&body_bytes),
213        signature,
214    };
215    if signed.verify(&public_key).is_err() {
216        warn!(agent_id = %agent_id, "auth: signature verification failed");
217        return Err(AuthError::Rejected);
218    }
219
220    // 6. Consume the nonce only now that the caller has proven who they are.
221    let fresh = NonceStore::new(pool.clone())
222        .check_and_consume(&agent_id, &nonce, NONCE_TTL_SECS)
223        .await
224        .map_err(|e| {
225            error!(error = %e, "auth: nonce store failed");
226            AuthError::Unavailable
227        })?;
228    if !fresh {
229        warn!(agent_id = %agent_id, "auth: nonce replay rejected");
230        return Err(AuthError::Malformed("nonce already used".to_string()));
231    }
232
233    parts.extensions.insert(AuthenticatedAgent(agent_id));
234    Ok(next.run(Request::from_parts(parts, Body::from(body_bytes))).await)
235}
