16/// Simple token bucket for rate limiting.
17///
18/// Tokens refill continuously at a fixed rate up to `max_tokens`.
19/// Each consumed request decrements the bucket; when empty, requests
20/// are denied until enough tokens have refilled.
21struct TokenBucket {
22    /// Current number of available tokens.
23    tokens: u64,
24    /// Maximum number of tokens the bucket can hold.
25    max_tokens: u64,
26    /// Time of the last token refill calculation.
27    last_refill: Instant,
28    /// Tokens added per second.
29    refill_rate: f64,
30}
31
32impl TokenBucket {
33    /// Create a new token bucket.
34    ///
35    /// `max_tokens` is both the initial token count and the capacity.
36    /// `window_secs` defines the time window over which all tokens
37    /// would be consumed at the refill rate.
38    fn new(max_tokens: u64, window_secs: u64) -> Self {
39        Self {
40            tokens: max_tokens,
41            max_tokens,
42            last_refill: Instant::now(),
43            refill_rate: max_tokens as f64 / window_secs as f64,
44        }
45    }
46
47    /// Attempt to consume one token, refilling based on elapsed time first.
48    ///
49    /// Returns `true` if a token was consumed, `false` if the bucket is empty.
50    fn try_consume(&mut self) -> bool {
51        let now = Instant::now();
52        let elapsed = now.duration_since(self.last_refill).as_secs_f64();
53        self.tokens =
54            (self.tokens as f64 + elapsed * self.refill_rate).min(self.max_tokens as f64) as u64;
55        self.last_refill = now;
56        if self.tokens > 0 {
57            self.tokens -= 1;
58            true
59        } else {
60            false
61        }
62    }
63}

65/// Shared rate limiter state with global and per-agent token buckets.
66///
67/// Cloning a [`RateLimiter`] is cheap — it shares the underlying buckets
68/// via `Arc<Mutex<_>>`. Each clone refers to the same rate-limit state.
69#[derive(Clone)]
70pub struct RateLimiter {
71    /// Global token bucket shared across all agents.
72    global: Arc<Mutex<TokenBucket>>,
73    /// Per-agent token buckets keyed by agent ID.
74    per_agent: Arc<Mutex<HashMap<String, TokenBucket>>>,
75    /// Maximum tokens per agent bucket.
76    agent_limit: u64,
77    /// Refill window in seconds.
78    window_secs: u64,
79}
80
81impl RateLimiter {
82    /// Create a new rate limiter with separate global and per-agent limits.
83    ///
84    /// `global_limit` caps total requests across all agents.
85    /// `agent_limit` caps requests from any single agent identified by
86    /// the `X-Agent-Id` header. Both use the same `window_secs` for refill timing.
87    pub fn new(global_limit: u64, agent_limit: u64, window_secs: u64) -> Self {
88        Self {
89            global: Arc::new(Mutex::new(TokenBucket::new(global_limit, window_secs))),
90            per_agent: Arc::new(Mutex::new(HashMap::new())),
91            agent_limit,
92            window_secs,
93        }
94    }
95
96    /// Check whether a request should be allowed.
97    ///
98    /// The global limit is always checked first. If an `agent_id` is provided,
99    /// the per-agent limit is also checked. Returns `false` if either bucket
100    /// is exhausted.
101    pub fn check(&self, agent_id: Option<&str>) -> bool {
102        // Check global limit first
103        if !self.global.lock().unwrap().try_consume() {
104            return false;
105        }
106        // Then per-agent limit
107        if let Some(id) = agent_id {
108            let mut map = self.per_agent.lock().unwrap();
109            let bucket = map
110                .entry(id.to_string())
111                .or_insert_with(|| TokenBucket::new(self.agent_limit, self.window_secs));
112            if !bucket.try_consume() {
113                return false;
114            }
115        }
116        true
117    }
118}

120/// Extract the agent ID from the `X-Agent-Id` request header.
121///
122/// Returns `None` when the header is absent or contains non-UTF-8 bytes.
123fn extract_agent_id(req: &Request<Body>) -> Option<String> {
124    req.headers().get("X-Agent-Id").and_then(|v| v.to_str().ok()).map(|s| s.to_string())
125}

127/// Axum middleware for rate limiting.
128///
129/// Rejects requests with HTTP 429 (Too Many Requests) when the global
130/// or per-agent token bucket is exhausted.  A `Retry-After` header is
131/// included with the configured window in seconds.
132pub async fn rate_limit(
133    Extension(rate_limiter): Extension<RateLimiter>,
134    req: Request<Body>,
135    next: Next,
136) -> Response {
137    let agent_id = extract_agent_id(&req);
138    if !rate_limiter.check(agent_id.as_deref()) {
139        let mut resp = Response::new(Body::from("rate limit exceeded"));
140        *resp.status_mut() = StatusCode::TOO_MANY_REQUESTS;
141        resp.headers_mut().insert(
142            header::RETRY_AFTER,
143            HeaderValue::from_str(&rate_limiter.window_secs.to_string()).unwrap(),
144        );
145        return resp;
146    }
147    next.run(req).await
148}
