1//! Idempotency middleware.
2//!
3//! Replays the stored response when a client retries a mutation with the same
4//! `Idempotency-Key`, so a retry after a timeout cannot execute it twice.
5//!
6//! Scope and rules:
7//!
8//! - The key is scoped to the **verified principal** (the
9//!   [`AuthenticatedAgent`] set by the authentication layer, which therefore
10//!   must run first) plus `METHOD /path`. Unauthenticated routes share the
11//!   `anonymous` principal. The `X-Agent-Id` header is never consulted: it is
12//!   attacker-controlled. The anonymous key namespace is therefore shared by
13//!   every caller: clients must choose unguessable keys (a random UUID), or a
14//!   stranger who guesses a key first makes the real caller's retry a 422.
15//! - Replaying a key with a **different body** is a client error (422), not a
16//!   silent replay of an unrelated response.
17//! - Only successful (`< 400`) responses to unsafe methods are stored, so a
18//!   failed attempt can be retried under the same key. `GET`/`HEAD`/`OPTIONS`
19//!   pass through untouched.
20//! - Request and response bodies are bounded; nothing is buffered unbounded.
21//!
22//! Known limitation: two *concurrent* requests with the same new key can both
23//! execute, because the key is stored after the handler finishes. Closing that
24//! needs a reserve-then-complete protocol.
25

43/// Request header carrying the client-chosen idempotency key.
44pub const X_IDEMPOTENCY_KEY: HeaderName = HeaderName::from_static("idempotency-key");
45/// Response header set on a replayed response.
46const X_IDEMPOTENT_REPLAYED: HeaderName = HeaderName::from_static("x-idempotent-replayed");
47
48const IDEMPOTENCY_TTL_SECS: i64 = 86400; // 24 hours
49const MAX_KEY_LEN: usize = 255;
50/// Largest response body that will be stored for replay.
51const MAX_CACHED_RESPONSE_BYTES: usize = 1024 * 1024;
52/// Principal used on routes that carry no authentication.
53const ANONYMOUS: &str = "anonymous";
54
55fn reject(status: StatusCode, message: &str) -> Response {
56    (status, Json(serde_json::json!({ "error": message }))).into_response()
57}

59/// Idempotency middleware.
60///
61/// For unsafe-method requests carrying an `Idempotency-Key` header:
62/// 1. look up key + principal + route
63/// 2. same body: return the stored response; different body: 422
64/// 3. no entry: run the request and store a successful response
65pub async fn enforce_idempotency(
66    State(pool): State<SharedPool>,
67    req: Request<Body>,
68    next: Next,
69) -> Response {
70    let Some(raw_key) = req.headers().get(&X_IDEMPOTENCY_KEY) else {
71        return next.run(req).await;
72    };
73    if matches!(*req.method(), Method::GET | Method::HEAD | Method::OPTIONS) {
74        return next.run(req).await;
75    }
76    let idempotency_key = match raw_key.to_str() {
77        Ok(k) if !k.is_empty() && k.len() <= MAX_KEY_LEN => k.to_string(),
78        _ => {
79            return reject(
80                StatusCode::BAD_REQUEST,
81                "Idempotency-Key must be 1-255 visible ASCII characters",
82            );
83        }
84    };
85
86    let principal_id = req
87        .extensions()
88        .get::<AuthenticatedAgent>()
89        .map_or_else(|| ANONYMOUS.to_string(), |agent| agent.0.clone());
90    // `req.uri()` has the nest prefix stripped; the original URI names the route.
91    let path = req
92        .extensions()
93        .get::<OriginalUri>()
94        .map_or_else(|| req.uri().path().to_string(), |original| original.0.path().to_string());
95    let route = format!("{} {}", req.method(), path);
96
97    let (parts, body) = req.into_parts();
98    let Ok(body_bytes) = to_bytes(body, MAX_BODY_BYTES).await else {
99        return reject(
100            StatusCode::PAYLOAD_TOO_LARGE,
101            &format!("request body exceeds {MAX_BODY_BYTES} bytes"),
102        );
103    };
104    let request_hash = hex::encode(Sha256::digest(&body_bytes));
105    let req = Request::from_parts(parts, Body::from(body_bytes));
106
107    let pool = pool.load();
108
109    let cached = sqlx::query(
110        r#"
111        SELECT request_hash, response_status, response_body
112        FROM idempotency_keys
113        WHERE principal_id = $1 AND route = $2 AND idempotency_key = $3
114          AND expires_at > NOW()
115        "#,
116    )
117    .bind(&principal_id)
118    .bind(&route)
119    .bind(&idempotency_key)
120    .fetch_optional(&pool)
121    .await;
122    let cached = match cached {
123        Ok(row) => row,
124        Err(e) => {
125            warn!(error = %e, "failed to check idempotency cache");
126            return reject(StatusCode::SERVICE_UNAVAILABLE, "idempotency store unavailable");
127        }
128    };
129
130    if let Some(row) = cached {
131        let stored_hash: String = row.try_get("request_hash").unwrap_or_default();
132        if stored_hash != request_hash {
133            return reject(
134                StatusCode::UNPROCESSABLE_ENTITY,
135                "Idempotency-Key was already used with a different request body",
136            );
137        }
138        info!(idempotency_key = %idempotency_key, "returning cached idempotent response");
139        let status_code: i32 = row.try_get("response_status").unwrap_or(200);
140        let status = u16::try_from(status_code)
141            .ok()
142            .and_then(|code| StatusCode::from_u16(code).ok())
143            .unwrap_or(StatusCode::OK);
144        let stored: Option<serde_json::Value> = row.try_get("response_body").ok().flatten();
145
146        let mut resp = match stored {
147            Some(json) => {
148                let mut r = Response::new(Body::from(json.to_string()));
149                r.headers_mut()
150                    .insert(header::CONTENT_TYPE, HeaderValue::from_static("application/json"));
151                r
152            }
153            None => Response::new(Body::empty()),
154        };
155        *resp.status_mut() = status;
156        resp.headers_mut().insert(X_IDEMPOTENT_REPLAYED, HeaderValue::from_static("true"));
157        return resp;
158    }
159
160    let resp = next.run(req).await;
161    if resp.status().as_u16() >= 400 {
162        return resp;
163    }
164
165    let (parts, body) = resp.into_parts();
166    let Ok(body_bytes) = to_bytes(body, MAX_CACHED_RESPONSE_BYTES).await else {
167        warn!("response too large to store for idempotency");
168        return reject(StatusCode::INTERNAL_SERVER_ERROR, "response too large");
169    };
170
171    let response_body: Option<serde_json::Value> =
172        if body_bytes.is_empty() { None } else { serde_json::from_slice(&body_bytes).ok() };
173    let expires_at = Utc::now() + chrono::Duration::seconds(IDEMPOTENCY_TTL_SECS);
174
175    let stored = sqlx::query(
176        r#"
177        INSERT INTO idempotency_keys
178            (principal_id, route, idempotency_key, request_hash, response_status, response_body, expires_at)
179        VALUES ($1, $2, $3, $4, $5, $6, $7)
180        ON CONFLICT (principal_id, route, idempotency_key) DO NOTHING
181        "#,
182    )
183    .bind(&principal_id)
184    .bind(&route)
185    .bind(&idempotency_key)
186    .bind(&request_hash)
187    .bind(i32::from(parts.status.as_u16()))
188    .bind(&response_body)
189    .bind(expires_at)
190    .execute(&pool)
191    .await;
192    if let Err(e) = stored {
193        warn!(error = %e, "failed to store idempotent response");
194    }
195
196    let mut new_resp = Response::from_parts(parts, Body::from(body_bytes));
197    if let Ok(value) = HeaderValue::from_str(&idempotency_key) {
198        new_resp.headers_mut().insert(HeaderName::from_static("x-idempotency-key"), value);
199    }
200    new_resp
201}
