1//! Periodic cleanup of tables that only ever grow.
2//!
3//! Nonces (replay protection) and idempotency keys carry an `expires_at`;
4//! nothing else removes them once it passes.
5
6use std::time::Duration;
7
8use ac_crypto::nonce::NonceStore;
9use ac_db::pool::SharedPool;
10use tracing::{error, info};
11
12/// How many rows one purge pass removed.
13#[derive(Debug, Default, Clone, Copy, PartialEq, Eq)]
14pub struct Purged {
15    /// Rows removed from `nonce_replay_cache`.
16    pub nonces: u64,
17    /// Rows removed from `idempotency_keys`.
18    pub idempotency_keys: u64,
19}
20
21/// Delete expired nonces and idempotency keys once.
22///
23/// # Errors
24///
25/// Returns the database error if either delete fails.
26pub async fn purge_expired(pool: &SharedPool) -> Result<Purged, sqlx::Error> {
27    let nonces = NonceStore::new(pool.clone()).purge_expired().await?;
28    let idempotency_keys = sqlx::query("DELETE FROM idempotency_keys WHERE expires_at < NOW()")
29        .execute(&pool.load())
30        .await?
31        .rows_affected();
32    Ok(Purged { nonces, idempotency_keys })
33}
34
35/// Run [`purge_expired`] every `interval` on the current Tokio runtime.
36///
37/// Long-lived servers call this once at startup. Short-lived (serverless)
38/// entry points have no background runtime and should call
39/// [`purge_expired`] from a scheduled invocation instead.
40pub fn spawn_purge_task(pool: SharedPool, interval: Duration) -> tokio::task::JoinHandle<()> {
41    tokio::spawn(async move {
42        let mut ticker = tokio::time::interval(interval);
43        loop {
44            ticker.tick().await;
45            match purge_expired(&pool).await {
46                Ok(p) if p != Purged::default() => info!(?p, "purged expired rows"),
47                Ok(_) => {}
48                Err(e) => error!(error = %e, "purge of expired rows failed"),
49            }
50        }
51    })
52}
