10/// Description of the request-signing contract, published as the
11/// `AgentSignature` security scheme.
12const AGENT_SIGNATURE_DESCRIPTION: &str = "\
13Ed25519 request signing (ACP/1). Send four headers on every protected request:\n\n\
14- `X-Agent-Id`: the registered agent id.\n\
15- `X-Agent-Timestamp`: an RFC 3339 UTC timestamp in whole seconds, for example \
16`2026-11-22T20:30:00Z`. It must be within 300 seconds of server time; integers (Unix \
17seconds) are rejected with 400.\n\
18- `X-Agent-Nonce`: a unique value, used once per agent (a reused nonce is rejected with \
19400 and remembered for at least 600 seconds).\n\
20- `X-Agent-Signature`: the hex Ed25519 signature of the UTF-8 string \
21`agent_id \\n timestamp \\n nonce \\n method \\n path_and_query \\n sha256_hex(body)` \
22(newline separated, no spaces). `path_and_query` is the request target exactly as sent, \
23including the query string, and `sha256_hex` is the lowercase hex SHA-256 of the raw body \
24bytes (empty body for GET).\n\n\
25Failures: 401 missing or failed authentication (also unknown or suspended agents), 400 \
26malformed, stale or replayed credentials, 413 body over 1 MiB. The optional \
27`Idempotency-Key` header makes a retried mutation return the stored response; reusing a \
28key with a different body is 422. Keys are scoped to the authenticated agent; on public \
29routes such as registration they are shared by all callers, so use an unguessable key (a \
30random UUID).";
31
32/// Registers the `AgentSignature` security scheme.
33struct SigningScheme;
34
35impl Modify for SigningScheme {
36    fn modify(&self, openapi: &mut utoipa::openapi::OpenApi) {
37        let components = openapi.components.get_or_insert_with(Default::default);
38        let mut scheme = ApiKeyValue::new("X-Agent-Signature");
39        scheme.description = Some(AGENT_SIGNATURE_DESCRIPTION.to_string());
40        components
41            .add_security_scheme("AgentSignature", SecurityScheme::ApiKey(ApiKey::Header(scheme)));
42    }
43}

45/// Root OpenAPI specification for the Agent Commons API.
46///
47/// This struct derives [`utoipa::OpenApi`] and collects all documented
48/// handler functions from every feature crate.  The resulting document
49/// includes:
50/// - All 19 API endpoint paths with request/response schemas
51/// - 7 tag groupings (system, registry, discovery, mailbox, tasks, validation, reputation)
52/// - API metadata (title, version, description, contact, license)
53/// - Server URL for the local development instance
54#[derive(OpenApi)]
55#[openapi(
56    modifiers(&SigningScheme),
57    paths(
58        crate::server::health_check,
59        crate::server::version_check,
60        ac_registry::handler::register_handler,
61        ac_registry::handler::get_agent_handler,
62        ac_registry::handler::update_card_handler,
63        ac_discovery::handler::search,
64        ac_mailbox::handler::send_message,
65        ac_mailbox::handler::get_messages,
66        ac_mailbox::handler::get_message,
67        ac_mailbox::handler::acknowledge_message,
68        ac_tasks::handler::create_task,
69        ac_tasks::handler::get_task,
70        ac_tasks::handler::accept_task,
71        ac_tasks::handler::reject_task,
72        ac_tasks::handler::submit_result,
73        ac_validation::handler::validate_task,
74        ac_validation::handler::get_validations,
75        ac_reputation::handler::get_reputation,
76        ac_reputation::handler::get_contributions,
77    ),
78    tags(
79        (name = "system", description = "Health and version endpoints"),
80        (name = "registry", description = "Agent registration and card management"),
81        (name = "discovery", description = "Capability-based agent search"),
82        (name = "mailbox", description = "Persistent message delivery"),
83        (name = "tasks", description = "Task lifecycle management"),
84        (name = "validation", description = "Peer validation and quorum"),
85        (name = "reputation", description = "Reputation scores and contribution stats"),
86    ),
87    info(
88        title = "Agent Commons API",

93    ),
94    servers(
95        (url = "http://localhost:3000", description = "Local development"),
96    ),
97)]
98pub struct ApiDoc;

100/// GET /api/openapi.json — the raw OpenAPI specification.
101///
102/// Returns the complete OpenAPI 3.0 document as JSON, suitable for
103/// importing into API clients, code generators, or documentation tools.
