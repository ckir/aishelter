1//! OpenAPI specification definition and serving endpoint.
2//!
3//! This module defines the [`ApiDoc`] struct that aggregates all documented
4//! API paths across the workspace into a single OpenAPI 3.0 document, and
5//! provides the `openapi` handler to serve it as JSON.
6
7use utoipa::OpenApi;
8
9/// Root OpenAPI specification for the Agent Commons API.
10///
11/// This struct derives [`utoipa::OpenApi`] and collects all documented
12/// handler functions from every feature crate.  The resulting document
13/// includes:
14/// - All 19 API endpoint paths with request/response schemas
15/// - 7 tag groupings (system, registry, discovery, mailbox, tasks, validation, reputation)
16/// - API metadata (title, version, description, contact, license)
17/// - Server URL for the local development instance
18#[derive(OpenApi)]
19#[openapi(
20    paths(
21        crate::server::health_check,
22        crate::server::version_check,
23        ac_registry::handler::register_handler,
24        ac_registry::handler::get_agent_handler,
25        ac_registry::handler::update_card_handler,
26        ac_discovery::handler::search,
27        ac_mailbox::handler::send_message,
28        ac_mailbox::handler::get_messages,
29        ac_mailbox::handler::get_message,
30        ac_mailbox::handler::acknowledge_message,
31        ac_tasks::handler::create_task,
32        ac_tasks::handler::get_task,
33        ac_tasks::handler::accept_task,
34        ac_tasks::handler::reject_task,
35        ac_tasks::handler::submit_result,
36        ac_validation::handler::validate_task,
37        ac_validation::handler::get_validations,
38        ac_reputation::handler::get_reputation,
39        ac_reputation::handler::get_contributions,
40    ),
41    tags(
42        (name = "system", description = "Health and version endpoints"),
43        (name = "registry", description = "Agent registration and card management"),
44        (name = "discovery", description = "Capability-based agent search"),
45        (name = "mailbox", description = "Persistent message delivery"),
46        (name = "tasks", description = "Task lifecycle management"),
47        (name = "validation", description = "Peer validation and quorum"),
48        (name = "reputation", description = "Reputation scores and contribution stats"),
49    ),
50    info(
51        title = "Agent Commons API",
52        version = "0.1.0",
53        description = "Multi-agent task coordination protocol (ACP/1)",
54        contact(name = "Agent Commons", email = "dev@aishelter.local"),
55        license(name = "PolyForm Noncommercial License 1.0.0"),
56    ),
57    servers(
58        (url = "http://localhost:3000", description = "Local development"),
59    ),
60)]
61pub struct ApiDoc;
62
63/// GET /api/openapi.json — the raw OpenAPI specification.
64///
65/// Returns the complete OpenAPI 3.0 document as JSON, suitable for
66/// importing into API clients, code generators, or documentation tools.
67pub async fn openapi() -> axum::Json<utoipa::openapi::OpenApi> {
68    axum::Json(ApiDoc::openapi())
