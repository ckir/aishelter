1//! Service Manifest type for `/.well-known/agent-service.json`.
2//!
3//! A Service Manifest describes a concrete service endpoint: its URL,
4//! schema version, capabilities, and authentication requirements.
5
6use chrono::{DateTime, Utc};
7use serde::{Deserialize, Serialize};
8
9use crate::Capability;
10
11/// Service Manifest — describes a discoverable service endpoint.
12///
13/// Published at `/.well-known/agent-service.json` and optionally stored
14/// in the database for directory-level caching and validation.
15#[derive(Debug, Clone, Serialize, Deserialize)]
16pub struct ServiceManifest {
17    /// Unique service identifier (e.g. `"svc_<UUID>"`).
18    pub service_id: String,
19    /// URL where the full manifest document can be fetched.
20    pub manifest_url: url::Url,
21    /// Schema version string (e.g. `"1.0"`).
22    pub schema_version: String,
23    /// Human-readable service name.
24    pub name: String,
25    /// Longer description of the service's purpose.
26    pub description: String,
27    /// List of capabilities this service exposes.
28    pub capabilities: Vec<Capability>,
29    /// Supported protocol identifiers (e.g. `"acp/1"`).
30    pub protocols: Vec<String>,
31    /// Whether this service requires authentication.
32    pub requires_auth: bool,
33    /// URL of the owner/agent that owns this service.
34    pub owner_url: Option<url::Url>,
35    /// Timestamp when the manifest was last fetched from the source URL.
36    pub fetched_at: Option<DateTime<Utc>>,
37    /// Timestamp when this record was last updated in the directory.
38    pub updated_at: Option<DateTime<Utc>>,
39}
