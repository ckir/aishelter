1//! Axum HTTP route handlers for task operations (§18).
2//!
3//! Exposes five REST endpoints:
4//! - `POST /` — create a new task contract
5//! - `GET /{id}` — retrieve task details
6//! - `POST /{id}/accept` — accept a task offer
7//! - `POST /{id}/reject` — reject a task offer
8//! - `POST /{id}/result` — submit a task result
9
10use ac_db::pool::SharedPool;
11use axum::{
12    Json, Router,
13    extract::{Path, State},
14    routing::{get, post},
15};
16use serde::Deserialize;
17
18use crate::service::TaskService;
19
20/// Build the task router with all five endpoints.
21pub fn routes(pool: SharedPool) -> Router {
22    Router::new()
23        .route("/", post(create_task))
24        .route("/{id}", get(get_task))
25        .route("/{id}/accept", post(accept_task))
26        .route("/{id}/reject", post(reject_task))
27        .route("/{id}/result", post(submit_result))
28        .with_state(pool)
29}
30
31/// Request body for [`create_task`].
32#[derive(Deserialize, utoipa::ToSchema)]
33pub struct CreateTaskRequest {
34    /// Agent ID of the task requester.
35    pub requester: String,
36    /// Capability required to execute the task (e.g. `"fact_verification"`).
37    pub capability: String,
38    /// Human-readable description of the work.
39    pub description: String,
40    /// JSON-encoded input data for the task executor.
41    pub input: serde_json::Value,
42    /// Optional deadline for task completion.
43    pub deadline: Option<chrono::DateTime<chrono::Utc>>,
44    /// Verification method: `"peer"`, `"requester"`, or `"deterministic"`.
45    pub verification_method: Option<String>,
46    /// Number of validator approvals required (default 2).
47    pub required_validators: Option<i32>,
48}
49
50/// Request body for [`accept_task`] and [`reject_task`].
51#[derive(Deserialize, utoipa::ToSchema)]
52pub struct TaskActionRequest {
53    /// Agent ID of the executor responding to the offer.
54    pub agent_id: String,
55}
56
57/// Request body for [`submit_result`].
58#[derive(Deserialize, utoipa::ToSchema)]
59pub struct SubmitResultRequest {
60    /// Agent ID of the executor submitting the result.
61    pub agent_id: String,
62    /// JSON-encoded result data.
63    pub result: serde_json::Value,
64    /// SHA-256 hash of the output for tamper-evident verification.
65    pub output_hash: String,
66}

68/// Create a new task contract.
69///
70/// The task starts in the `CREATED` state.  The requester is recorded,
71/// and the capability, input data, and verification parameters are stored
72/// (§18).
73#[utoipa::path(
74    post,
75    path = "/v1/tasks",
76    tag = "tasks",
77    request_body = CreateTaskRequest,
78    responses(
79        (status = 200, description = "Task created", body = serde_json::Value),
80    ),
81)]
82pub async fn create_task(
83    State(pool): State<SharedPool>,
84    Json(req): Json<CreateTaskRequest>,
85) -> Result<Json<serde_json::Value>, ac_types::error::AcError> {
86    let pool = pool.load();
87    let service = TaskService::new(pool);
88    let verification_method = req.verification_method.unwrap_or_else(|| "peer".to_string());
89    let required_validators = req.required_validators.unwrap_or(2);
90
91    let task_id = service
92        .create_task(
93            &req.requester,
94            &req.capability,
95            &req.description,
96            req.input,
97            req.deadline,
98            &verification_method,
99            required_validators,
100        )
101        .await?;
102
103    Ok(Json(serde_json::json!({ "task_id": task_id, "status": "CREATED" })))
104}

118pub async fn get_task(
119    State(pool): State<SharedPool>,
120    Path(id): Path<String>,
121) -> Result<Json<serde_json::Value>, ac_types::error::AcError> {
122    let pool = pool.load();
123    let service = TaskService::new(pool);
124    let task = service.get_task(&id).await?;
125    Ok(Json(serde_json::json!({ "data": task })))
126}

142pub async fn accept_task(
143    State(pool): State<SharedPool>,
144    Path(id): Path<String>,
145    Json(req): Json<TaskActionRequest>,
146) -> Result<Json<serde_json::Value>, ac_types::error::AcError> {
147    let pool = pool.load();
148    let service = TaskService::new(pool);
149    service.accept_task(&id, &req.agent_id).await?;
150    Ok(Json(serde_json::json!({ "status": "accepted" })))
151}
