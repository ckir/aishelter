1//! Task state machine: valid transitions between [`ac_types::task::TaskStatus`] variants.
2//!
3//! The task lifecycle (§18) is a directed graph.  This module provides the
4//! sole authority on which transitions are legal, preventing invalid state
5//! changes at both the API and service layers.
6//!
7//! # Valid transitions
8//!
9//! ```text
10//! CREATED → OFFERED → ACCEPTED → RUNNING → SUBMITTED → VERIFYING → VERIFIED → CLOSED
11//!                   → REJECTED                                         → REJECTED → CLOSED
12//!                                                                  → DISPUTED → CLOSED
13//! ```
14
15use ac_types::task::TaskStatus;
16
17/// Check whether a transition from `from` to `to` is valid.
18///
19/// // Returns true only for transitions defined in the task protocol (§18).
20pub fn is_valid_transition(from: TaskStatus, to: TaskStatus) -> bool {
21    matches!(
22        (from, to),
23        // Normal forward path.
24        (TaskStatus::Created, TaskStatus::Offered)
25            | (TaskStatus::Offered, TaskStatus::Accepted)
26            | (TaskStatus::Accepted, TaskStatus::Running)
27            | (TaskStatus::Running, TaskStatus::Submitted)
28            | (TaskStatus::Submitted, TaskStatus::Verifying)
29            | (TaskStatus::Verifying, TaskStatus::Verified)
30            // Rejection branch from OFFERED or VERIFYING.
31            | (TaskStatus::Offered, TaskStatus::Rejected)
32            | (TaskStatus::Verifying, TaskStatus::Rejected)
33            // Dispute branch from VERIFYING.
34            | (TaskStatus::Verifying, TaskStatus::Disputed)
35            // Closure: all terminal states lead to CLOSED.
36            | (TaskStatus::Verified, TaskStatus::Closed)
37            | (TaskStatus::Rejected, TaskStatus::Closed)
38            | (TaskStatus::Disputed, TaskStatus::Closed)
39    )
40}
