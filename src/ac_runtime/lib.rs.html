1//! Runtime adapter traits for serverless deployments.
2//!
3//! This crate defines the `HttpAdapter` and `DbAdapter` traits that
4//! decouple the ac-server application from platform-specific HTTP
5//! handling and database connection logic.
6//!
7//! Each serverless platform (AWS Lambda, Cloudflare Workers, Google
8//! Cloud Run) has its own event loop and lifecycle. These traits let
9//! ac-server hand off a pre-built `axum::Router` to whichever adapter
10//! the deployment target requires, without conditional compilation.
11
12use axum::Router;
13use sqlx::PgPool;
14
15/// Result type alias for adapter operations.
16///
17/// Used by all `HttpAdapter` and `DbAdapter` methods to surface errors
18/// uniformly regardless of platform. Backed by `anyhow::Error` so
19/// implementations can use `?` with any error type.
20pub type Result<T> = std::result::Result<T, anyhow::Error>;
21
22/// Abstracts platform-specific HTTP request/response handling.
23///
24/// Each implementation wraps the axum `Router` in a platform-specific
25/// event loop — for Lambda this invokes `lambda_http::run`, for Workers
26/// it calls into `worker::WorkerService`, and for Cloud Run it binds
27/// to a TCP socket.
28///
29/// This trait is NOT dyn-object-safe — the `impl Future` return type prevents
30/// `Box<dyn HttpAdapter>`. Use generics (`fn serve_with<T: HttpAdapter>(...)`)
31/// to accept adapters polymorphically.
32pub trait HttpAdapter: Send + Sync + 'static {
33    /// Start serving requests through the given router.
34    ///
35    /// Blocks until the platform signals shutdown (SIGTERM, Lambda
36    /// runtime exit, etc.) or an unrecoverable error occurs. Returns
37    /// `Ok(())` on clean shutdown, `Err` otherwise.
38    fn serve(self, router: Router) -> impl std::future::Future<Output = Result<()>> + Send;
39}
40
41/// Abstracts platform-specific database connection creation.
42///
43/// For Lambda and Cloud Run the implementation returns a
44/// `sqlx::PgPool` backed by a direct TCP connection. For Workers the
45/// implementation may return an HTTP gateway client that proxies SQL
46/// over HTTPS to a pooled edge service.
47///
48/// The associated `Pool` type lets each platform choose the most
49/// appropriate connection representation while sharing the same trait
50/// interface.
51pub trait DbAdapter: Send + Sync + 'static {
52    /// The connection pool type this adapter produces.
53    ///
54    /// Must be cloneable and thread-safe so that request handlers can
55    /// share a single pool across concurrent invocations.
56    type Pool: Clone + Send + Sync + 'static;
57
58    /// Create a new connection pool from the given connection string.
59    ///
60    /// The connection string is expected to be a PostgreSQL URI of the
61    /// form `postgresql://user:pass@host:port/dbname`. Implementations
62    /// choose their own pooling strategy (max connections, timeouts,
63    /// TLS) based on platform constraints.
64    fn create_pool(
65        connection_string: &str,
66    ) -> impl std::future::Future<Output = Result<Self::Pool>> + Send;
67}
68
69/// A `DbAdapter` that returns a standard `sqlx::PgPool`.
70///
71/// Used by Lambda and Cloud Run deployments where a direct TCP
72/// connection to PostgreSQL is available. Configures a conservative
73/// `max_connections` of 10 to stay within serverless connection limits.
74pub struct PgDbAdapter;
75
76impl DbAdapter for PgDbAdapter {
77    type Pool = PgPool;
78
79    /// Create a `PgPool` from the connection string.
80    ///
81    /// Configures a maximum of 10 concurrent connections, suitable for
82    /// most serverless workloads. Override per-deployment by providing
83    /// a custom `DbAdapter` implementation.
84    async fn create_pool(connection_string: &str) -> Result<Self::Pool> {
85        // Use a conservative max_connections for serverless environments
86        // where connection limits are typically low.
87        let pool = sqlx::postgres::PgPoolOptions::new()
88            .max_connections(10)
89            .connect(connection_string)
90            .await?;
91        Ok(pool)
92    }
93}

95// Platform-specific adapter modules, compiled only when their feature
96// flags are enabled. This keeps the dependency tree lean — a Cloud Run
97// deployment doesn't pull in lambda_http, for example.
98
99/// Adapter implementations for specific deployment platforms.
100///
101/// Each sub-module provides one or more `HttpAdapter` implementations
102/// gated behind a feature flag. Enable only the adapter(s) needed for
103/// the target deployment.
104#[cfg(any(feature = "lambda", feature = "cloudrun", feature = "workers"))]
105pub mod adapters;
106
107/// Re-export the AWS Lambda HTTP adapter when the `lambda` feature is enabled.
108///
109/// Allows users to write `use ac_runtime::LambdaHttpAdapter` without
110/// navigating through the adapters module path.
111#[cfg(feature = "lambda")]
112pub use adapters::lambda::LambdaHttpAdapter;
113
114/// Re-export the Google Cloud Run HTTP adapter when the `cloudrun` feature is enabled.
115///
116/// Allows users to write `use ac_runtime::CloudRunHttpAdapter` without
117/// navigating through the adapters module path.
118#[cfg(feature = "cloudrun")]
119pub use adapters::cloudrun::CloudRunHttpAdapter;
120
121/// Re-export the Cloudflare Workers HTTP adapter when the `workers` feature is enabled.
122///
123/// Allows users to write `use ac_runtime::WorkersHttpAdapter` without
124/// navigating through the adapters module path.
125#[cfg(feature = "workers")]
126pub use adapters::workers::WorkersHttpAdapter;
