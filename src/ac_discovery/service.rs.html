1//! Discovery service layer.
2//!
3//! The [`DiscoveryService`] provides capability-based agent search with
4//! optional filtering by reliability score and status (§13-14).  Results
5//! are joined from the `agents`, `agent_capabilities`, and
6//! `reputation_snapshots` tables, ordered by reliability descending.
7
8use ac_types::error::AcError;
9use sqlx::PgPool;
10
11/// A single result from a discovery search query.
12///
13/// Each result includes the agent's ID, the matched capability name and
14/// version, optional profile name, and the agent's current reliability
15/// score and confidence from the reputation snapshot.
16#[derive(Debug, Clone, serde::Serialize)]
17pub struct DiscoveryResult {
18    /// The agent's unique identifier.
19    pub agent_id: String,
20    /// The matched capability name.
21    pub capability: String,
22    /// Version string of the matched capability.
23    pub capability_version: String,
24    /// The agent's human-readable profile name (if set).
25    pub profile_name: Option<String>,
26    /// Reliability score from the reputation snapshot (0.0..1.0).
27    pub reliability: Option<f64>,
28    /// Confidence in the reliability score (0.0..1.0).
29    pub reliability_confidence: Option<f64>,
30}
31
32/// Type alias for the raw database row tuple returned by the discovery query.
33type DiscoveryRow = (String, String, String, Option<String>, Option<f64>, Option<f64>);
34
35/// Query parameters for agent discovery search (§13).
36///
37/// This internal struct holds the parsed search criteria.  When `capability`
38/// is `Some`, the query joins with `agent_capabilities` to filter by name.
39/// When `min_reliability` is set, results below that threshold are removed
40/// post-query.
41#[derive(Debug, Default)]
42pub struct SearchQuery {
43    /// Filter by capability name.
44    pub capability: Option<String>,
45    /// Minimum reliability score (0.0..1.0).
46    pub min_reliability: Option<f64>,
47    /// Protocol filter (e.g. `"acp/1"`).
48    pub protocol: Option<String>,
49    /// Agent status filter (default: `"ACTIVE"`).
50    pub status: Option<String>,
51    /// Maximum number of results (default: 20).
52    pub limit: i64,
53}
54
55/// The discovery service handles agent search and ranking.
56///
57/// Wraps a [`PgPool`] and provides [`search_agents`] for capability-based
58/// discovery with optional reliability filtering.
59pub struct DiscoveryService {
60    /// The PostgreSQL connection pool.
61    pool: PgPool,
62}
63
64impl DiscoveryService {
65    /// Create a new [`DiscoveryService`] from an existing connection pool.
66    pub fn new(pool: PgPool) -> Self {
67        Self { pool }
68    }
69
70    /// Search for agents by capability and optional filters.
71    ///
72    /// When a `capability` is provided, the query joins `agents` with
73    /// `agent_capabilities` and filters by capability name.  Without a
74    /// capability, it returns all agents with at least one capability.
75    /// Results are ordered by reliability score descending (NULLs last)
76    /// and limited to the specified count.
77    ///
78    /// After fetching, `min_reliability` is applied as an in-memory filter
79    /// to remove agents below the threshold.
80    pub async fn search_agents(
81        &self,
82        query: &SearchQuery,
83    ) -> Result<Vec<DiscoveryResult>, AcError> {
84        // Use a sensible default limit if zero or negative.
85        let limit = if query.limit > 0 { query.limit } else { 20 };
86        // Default to ACTIVE status when no explicit status filter is given.
87        let status = query.status.as_deref().unwrap_or("ACTIVE");
88
89        // Execute the query — with or without capability filter.
90        let results: Vec<DiscoveryRow> = if let Some(ref cap) = query.capability {
91            // Capability-filtered query: join with agent_capabilities.
92            sqlx::query_as(
93                r#"
94                SELECT
95                    a.agent_id,
96                    ac.capability_name,
97                    ac.capability_version,
98                    a.profile_name,
99                    rs.reliability,
100                    rs.reliability_confidence
101                FROM agents a
102                JOIN agent_capabilities ac ON a.agent_id = ac.agent_id
103                LEFT JOIN reputation_snapshots rs ON a.agent_id = rs.agent_id
104                WHERE ac.capability_name = $1
105                  AND a.status = $2
106                ORDER BY rs.reliability DESC NULLS LAST
107                LIMIT $3
108                "#,
109            )
110            .bind(cap)
111            .bind(status)
112            .bind(limit)
113            .fetch_all(&self.pool)
114            .await
115            .map_err(|e| AcError::Database(e.to_string()))?
116        } else {
117            // Unfiltered query: return all agents with capabilities.
118            sqlx::query_as(
119                r#"
120                SELECT
121                    a.agent_id,
122                    ac.capability_name,
123                    ac.capability_version,
124                    a.profile_name,
125                    rs.reliability,
126                    rs.reliability_confidence
127                FROM agents a
128                JOIN agent_capabilities ac ON a.agent_id = ac.agent_id
129                LEFT JOIN reputation_snapshots rs ON a.agent_id = rs.agent_id
130                WHERE a.status = $1
131                ORDER BY rs.reliability DESC NULLS LAST
132                LIMIT $2
133                "#,
134            )
135            .bind(status)
136            .bind(limit)
137            .fetch_all(&self.pool)
138            .await
139            .map_err(|e| AcError::Database(e.to_string()))?
140        };
141
142        // Map database rows to typed DiscoveryResult structs.
143        let mut discovery_results: Vec<DiscoveryResult> = results
144            .into_iter()
145            .map(|(agent_id, cap_name, cap_ver, profile, reliability, rel_conf)| DiscoveryResult {
146                agent_id,
147                capability: cap_name,
148                capability_version: cap_ver,
149                profile_name: profile,
150                reliability,
151                reliability_confidence: rel_conf,
152            })
153            .collect();
154
155        // Apply post-query reliability threshold filter.
156        if let Some(min_rel) = query.min_reliability {
157            discovery_results.retain(|r| r.reliability.is_some_and(|v| v >= min_rel));
158        }
159
160        Ok(discovery_results)
161    }
162}
