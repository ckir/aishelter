1//! MCP tool declarations for the Agent Commons adapter.
2//!
3//! Each tool exposes a name, description, and JSON Schema input definition.
4//! These are returned by the `tools/list` MCP method and used by the
5//! `tools/call` dispatcher to route to the corresponding ACP/1 operation.
6
7use serde_json::{Value, json};
8
9/// Metadata for a single MCP tool.
10#[derive(Debug, Clone, serde::Serialize)]
11pub struct McpTool {
12    /// Tool name used in `tools/call` requests.
13    pub name: &'static str,
14    /// Human-readable description of what the tool does.
15    pub description: &'static str,
16    /// JSON Schema describing the `arguments` object for `tools/call`.
17    pub input_schema: Value,
18}
19
