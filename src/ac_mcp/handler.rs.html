2//!
3//! Translates incoming MCP JSON-RPC requests into ACP/1 internal service
4//! calls and maps responses back to JSON-RPC 2.0 format.  The single
5//! `POST /mcp` endpoint accepts all MCP methods.

19// ---------------------------------------------------------------------------
20// JSON-RPC 2.0 envelope types

23/// Incoming JSON-RPC 2.0 request.
24#[derive(Debug, Deserialize)]
25pub struct JsonRpcRequest {
26    /// Always `"2.0"`.
27    pub jsonrpc: String,
28    /// Method name (e.g. `"tools/list"`, `"tools/call"`).
29    pub method: String,
30    /// Optional method parameters.
31    pub params: Option<Value>,
32    /// Request identifier (null for notifications).
33    pub id: Option<Value>,
34}
35
36/// Successful JSON-RPC 2.0 response.
37#[derive(Debug, Serialize)]
38pub struct JsonRpcSuccess<T> {
39    pub jsonrpc: &'static str,
40    pub result: T,
41    #[serde(skip_serializing_if = "Option::is_none")]
42    pub id: Option<Value>,
43}
44
45/// JSON-RPC 2.0 error response.
46#[derive(Debug, Serialize)]
47pub struct JsonRpcError {
48    pub jsonrpc: &'static str,
49    pub error: RpcErrorBody,
50    #[serde(skip_serializing_if = "Option::is_none")]
51    pub id: Option<Value>,
52}
53
54/// Unified JSON-RPC 2.0 response (success or error).
55#[derive(Debug, Serialize)]
56#[serde(untagged)]
57pub enum JsonRpcResponse {
58    Success(JsonRpcSuccess<Value>),
59    Error(JsonRpcError),
60}
61
62impl From<JsonRpcSuccess<Value>> for JsonRpcResponse {
63    fn from(s: JsonRpcSuccess<Value>) -> Self {
64        JsonRpcResponse::Success(s)
65    }
66}
67
68impl From<JsonRpcError> for JsonRpcResponse {
69    fn from(e: JsonRpcError) -> Self {
70        JsonRpcResponse::Error(e)
71    }
72}
73
74/// The `error` object inside a JSON-RPC error response.
75#[derive(Debug, Serialize)]
76pub struct RpcErrorBody {
77    pub code: i64,
78    pub message: String,
79    #[serde(skip_serializing_if = "Option::is_none")]
80    pub data: Option<Value>,
81}

151/// JSON-RPC error codes used by this adapter.
152mod codes {
153    #[allow(dead_code)]
154    pub const PARSE_ERROR: i64 = -32700;
155    pub const INVALID_REQUEST: i64 = -32600;
156    pub const METHOD_NOT_FOUND: i64 = -32601;
157    #[allow(dead_code)]
158    pub const INTERNAL_ERROR: i64 = -32603;
159    /// Application-level error (mapped from [`ac_types::error::AcError`]).
160    pub const APPLICATION_ERROR: i64 = -32000;
161}
162
163fn rpc_error(code: i64, message: impl Into<String>, id: Option<Value>) -> Json<JsonRpcResponse> {
164    Json(JsonRpcResponse::Error(JsonRpcError {
165        jsonrpc: "2.0",
166        error: RpcErrorBody { code, message: message.into(), data: None },
167        id,
168    }))
169}
170
171fn success(result: Value, id: Option<Value>) -> Json<JsonRpcResponse> {
172    Json(JsonRpcResponse::Success(JsonRpcSuccess { jsonrpc: "2.0", result, id }))
173}

179/// Execute a named `tools/call` request against the ACP/1 service layer.
180async fn dispatch_call(
181    pool: &SharedPool,
182    name: &str,
183    arguments: Option<Value>,
184) -> Result<Value, ac_types::error::AcError> {
185    match name {
186        "search_agents" => call_search_agents(pool, arguments).await,
187        "register_agent" => call_register_agent(pool, arguments).await,
188        "send_message" => call_send_message(pool, arguments).await,
189        "create_task" => call_create_task(pool, arguments).await,
190        "submit_result" => call_submit_result(pool, arguments).await,
191        "validate" => call_validate(pool, arguments).await,
192        other => Err(ac_types::error::AcError::Internal(format!("unknown MCP tool: {other}"))),
193    }
194}
195
196async fn call_search_agents(
197    pool: &SharedPool,
198    args: Option<Value>,
199) -> Result<Value, ac_types::error::AcError> {
200    let args: SearchAgentsArgs = args
201        .map(serde_json::from_value)
202        .transpose()
203        .map_err(|e| ac_types::error::AcError::Internal(format!("bad search_agents args: {e}")))?
204        .unwrap_or(SearchAgentsArgs {
205            capability: None,
206            min_reliability: None,
207            protocol: None,
208            status: None,
209            limit: None,
210        });
211
212    let db_pool = pool.load();
213    let service = DiscoveryService::new(db_pool);
214    let query = SearchQuery {
215        capability: args.capability,
216        min_reliability: args.min_reliability,
217        protocol: args.protocol,
218        status: args.status,
219        limit: args.limit.unwrap_or(20),
220    };
221
222    let results = service.search_agents(&query).await?;
223    Ok(serde_json::json!({
224        "results": results,
225        "count": results.len(),
226        "protocol": "acp/1",
227    }))
228}

230async fn call_register_agent(
231    pool: &SharedPool,
232    args: Option<Value>,
233) -> Result<Value, ac_types::error::AcError> {

238        .ok_or_else(|| {
239            ac_types::error::AcError::Internal("register_agent requires arguments".to_string())
240        })?;
241
242    let db_pool = pool.load();
243    let service = RegistryService::new(db_pool);
244    let agent = service
245        .register_agent(
246            &args.agent_id,
247            &args.public_key,
248            args.profile_name,
249            args.profile_description,
250        )
251        .await?;
252
253    Ok(serde_json::json!({
254        "agent_id": agent.agent_id,
255        "status": agent.status,
256        "protocol": "acp/1",
257    }))
258}

260async fn call_send_message(
261    pool: &SharedPool,
262    args: Option<Value>,
263) -> Result<Value, ac_types::error::AcError> {
264    let args: SendMessageArgs = args
265        .map(serde_json::from_value)
266        .transpose()
267        .map_err(|e| ac_types::error::AcError::Internal(format!("bad send_message args: {e}")))?
268        .ok_or_else(|| {
269            ac_types::error::AcError::Internal("send_message requires arguments".to_string())
270        })?;
271
272    let expires_at = args
273        .expires_at
274        .and_then(|s| chrono::DateTime::parse_from_rfc3339(&s).ok())
275        .map(|dt| dt.with_timezone(&chrono::Utc));
276
277    let db_pool = pool.load();
278    let service = MailboxService::new(db_pool);
279    let message_id = service
280        .send_message(
281            &args.from_agent_id,
282            &args.to_agent_id,
283            &args.message_type,
284            args.payload,
285            expires_at,
286        )
287        .await?;
288
289    Ok(serde_json::json!({
290        "status": "sent",
291        "message_id": message_id,
292    }))
293}

295async fn call_create_task(
296    pool: &SharedPool,
297    args: Option<Value>,
298) -> Result<Value, ac_types::error::AcError> {
299    let args: CreateTaskArgs = args
300        .map(serde_json::from_value)
301        .transpose()
302        .map_err(|e| ac_types::error::AcError::Internal(format!("bad create_task args: {e}")))?
303        .ok_or_else(|| {
304            ac_types::error::AcError::Internal("create_task requires arguments".to_string())
305        })?;
306
307    let deadline = args
308        .deadline
309        .and_then(|s| chrono::DateTime::parse_from_rfc3339(&s).ok())
310        .map(|dt| dt.with_timezone(&chrono::Utc));
311
312    let verification_method = args.verification_method.unwrap_or_else(|| "peer".to_string());
313    let required_validators = args.required_validators.unwrap_or(2);
314
315    let db_pool = pool.load();
316    let service = TaskService::new(db_pool);
317    let task_id = service
318        .create_task(
319            &args.requester,
320            &args.capability,
321            &args.description,
322            args.input,
323            deadline,
324            &verification_method,
325            required_validators,
326        )
327        .await?;
328
329    Ok(serde_json::json!({
330        "task_id": task_id,
331        "status": "CREATED",
332    }))
333}

356async fn call_validate(
357    pool: &SharedPool,
358    args: Option<Value>,
359) -> Result<Value, ac_types::error::AcError> {
360    let args: ValidateArgs = args
361        .map(serde_json::from_value)
362        .transpose()
363        .map_err(|e| ac_types::error::AcError::Internal(format!("bad validate args: {e}")))?
364        .ok_or_else(|| {
365            ac_types::error::AcError::Internal("validate requires arguments".to_string())
366        })?;
367
368    let db_pool = pool.load();
369    let service = ValidationService::new(db_pool);
370    let decision = service
371        .validate_task(&args.task_id, &args.validator_id, &args.decision, args.reasoning.as_deref())
372        .await?;
373
374    let decision_str = match decision {
375        ac_validation::quorum::QuorumDecision::Verified => "verified",
376        ac_validation::quorum::QuorumDecision::Rejected => "rejected",
377        ac_validation::quorum::QuorumDecision::Disputed => "disputed",
378        ac_validation::quorum::QuorumDecision::Pending => "pending",
379    };
380
381    Ok(serde_json::json!({
382        "decision": decision_str,
383    }))
384}

390/// POST /mcp — JSON-RPC 2.0 endpoint.
391///
392/// Accepts MCP `tools/list` and `tools/call` methods and translates them
393/// into ACP/1 service calls.  Unsupported methods return JSON-RPC
394/// `Method not found`.
395async fn mcp_handler(
396    State(pool): State<SharedPool>,
397    Json(req): Json<JsonRpcRequest>,
398) -> Json<JsonRpcResponse> {
399    let id = req.id.clone();
400
401    match req.method.as_str() {
402        "tools/list" => {
403            let tools: Vec<Value> = list_tools()
404                .into_iter()
405                .map(|t| {
406                    serde_json::json!({
407                        "name": t.name,
408                        "description": t.description,
409                        "inputSchema": t.input_schema,
410                    })
411                })
412                .collect();
413            success(serde_json::json!({ "tools": tools }), id)
414        }
415
416        "tools/call" => {
417            let params: ToolsCallParams = match req.params.map(serde_json::from_value) {
418                Some(Ok(p)) => p,
419                Some(Err(e)) => {
420                    return rpc_error(codes::INVALID_REQUEST, e.to_string(), id);
421                }
422                None => {
423                    return rpc_error(codes::INVALID_REQUEST, "tools/call requires params", id);
424                }
425            };
426
427            match dispatch_call(&pool, &params.name, params.arguments).await {
428                Ok(result) => success(
429                    serde_json::json!({
430                        "content": [{ "type": "text", "text": result.to_string() }],
431                    }),
432                    id,
433                ),
434                Err(e) => rpc_error(codes::APPLICATION_ERROR, e.to_string(), id),
435            }
436        }
437
438        other => rpc_error(codes::METHOD_NOT_FOUND, format!("method not found: {other}"), id),
439    }
440}

442/// Build the MCP JSON-RPC router.
443///
444/// The router exposes a single `POST /mcp` endpoint.  Callers send
445/// JSON-RPC 2.0 envelopes and receive MCP-compatible responses.
446pub fn mcp_router(pool: SharedPool) -> Router {
447    Router::new().route("/mcp", post(mcp_handler)).with_state(pool)
