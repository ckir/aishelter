1//! Quorum computation and decision types.
2//!
3//! Defines the [`QuorumDecision`] enum and [`QuorumResult`] struct that
4//! determine whether a task result passes, fails, or enters a dispute
5//! based on validator votes (§20-22).
6
7use ac_types::task::TaskStatus;
8use serde::{Deserialize, Serialize};
9
10/// The outcome of a quorum vote on a task result.
11///
12/// The decision is determined by comparing approval/rejection counts
13/// against the task's required validator threshold.
14#[derive(Debug, Clone, Copy, Serialize, Deserialize)]
15#[serde(rename_all = "SCREAMING_SNAKE_CASE")]
16pub enum QuorumDecision {
17    /// The result passed validation (approvals >= required).
18    Verified,
19    /// The result failed validation (rejections >= required).
20    Rejected,
21    /// Validators disagreed with both approvals and rejections present.
22    Disputed,
23    /// Not enough votes to reach a conclusion yet.
24    Pending,
25}
26
27/// Aggregated quorum result from validator votes.
28///
29/// This struct holds the raw counts and provides the [`decision`](QuorumResult::decision)
30/// and [`next_status`](QuorumResult::next_status) methods that determine
31/// the task's fate.
32pub struct QuorumResult {
33    /// Number of validators who approved the result.
34    pub approved: usize,
35    /// Number of validators who rejected the result.
36    pub rejected: usize,
37    /// Number of approvals required to pass validation.
38    pub required: usize,
39}
40
41impl QuorumResult {
42    /// Compute the quorum decision from the current vote counts.
43    ///
44    /// # Decision logic
45    ///
46    /// 1. If `approved >= required` → [`Verified`](QuorumDecision::Verified)
47    /// 2. If `rejected >= required` → [`Rejected`](QuorumDecision::Rejected)
48    /// 3. If total votes >= required AND both sides have votes → [`Disputed`](QuorumDecision::Disputed)
49    /// 4. Otherwise → [`Pending`](QuorumDecision::Pending)
50    pub fn decision(&self) -> QuorumDecision {
51        if self.approved >= self.required {
52            QuorumDecision::Verified
53        } else if self.rejected >= self.required {
54            QuorumDecision::Rejected
55        } else if (self.approved + self.rejected) >= self.required
56            && self.approved > 0
57            && self.rejected > 0
58        {
59            QuorumDecision::Disputed
60        } else {
61            QuorumDecision::Pending
62        }
63    }
64
65    /// Map the quorum decision to the corresponding task status.
66    ///
67    /// Used to transition the task to its final state once quorum is reached.
68    pub fn next_status(&self) -> TaskStatus {
69        match self.decision() {
70            QuorumDecision::Verified => TaskStatus::Verified,
71            QuorumDecision::Rejected => TaskStatus::Rejected,
72            QuorumDecision::Disputed => TaskStatus::Disputed,
73            QuorumDecision::Pending => TaskStatus::Verifying,
74        }
75    }
76}
77
78/// Compute a quorum decision from raw vote counts.
79///
80/// This is a convenience function that constructs a [`QuorumResult`]
81/// and returns its [`decision`](QuorumResult::decision).
82pub fn compute_quorum(required: usize, approvals: usize, rejections: usize) -> QuorumDecision {
83    let result = QuorumResult { approved: approvals, rejected: rejections, required };
84    result.decision()
85}
