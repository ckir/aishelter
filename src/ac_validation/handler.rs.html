1//! Axum route handlers for peer validation endpoints.
2//!
3//! Exposes two HTTP endpoints for the validation workflow (§20-22):
4//! - `POST /{task_id}/validate` — submit a validation decision
5//! - `GET /{task_id}` — retrieve all validation decisions for a task
6
7use ac_db::pool::SharedPool;
8use axum::{
9    Json, Router,
10    extract::{Path, State},
11    routing::{get, post},
12};
13use serde::{Deserialize, Serialize};
14
15use crate::quorum::QuorumDecision;
16use crate::service::ValidationService;
17
18/// Build the validation router with the given database pool.
19pub fn routes(pool: SharedPool) -> Router {
20    Router::new()
21        .route("/{task_id}/validate", post(validate_task))
22        .route("/{task_id}", get(get_validations))
23        .with_state(pool)
24}
25
26/// Request body for submitting a validation decision.
27#[derive(Deserialize, utoipa::ToSchema)]
28pub struct ValidateRequest {
29    /// The validator agent's ID.
30    pub validator_id: String,
31    /// The decision: `"approve"` or `"reject"`.
32    pub decision: String,
33    /// Optional reasoning for the decision.
34    pub reasoning: Option<String>,
35}
36
37/// Response body containing the quorum decision.
38#[derive(Serialize, utoipa::ToSchema)]
39pub struct ValidateResponse {
40    /// The quorum result: `"verified"`, `"rejected"`, `"disputed"`, or `"pending"`.
41    pub decision: String,
42}
43
44/// Submit a validation decision for a task.
45///
46/// Records the validator's vote and computes the quorum decision.
47/// If the quorum threshold is met, the task status is updated accordingly.
48#[utoipa::path(
49    post,
50    path = "/v1/validations/{task_id}/validate",
51    tag = "validation",
52    request_body = ValidateRequest,
53    params(
54        ("task_id" = String, Path, description = "Task ID"),
55    ),
56    responses(
57        (status = 200, description = "Quorum decision", body = ValidateResponse,
58         example = json!({"decision": "verified"})),
59        (status = 400, description = "Invalid request"),
60        (status = 404, description = "Task not found"),
61    ),
62)]
63pub async fn validate_task(
64    State(pool): State<SharedPool>,
65    Path(task_id): Path<String>,
66    Json(req): Json<ValidateRequest>,
67) -> Result<Json<ValidateResponse>, ac_types::error::AcError> {
68    let pool = pool.load();
69    let service = ValidationService::new(pool);
70    let decision = service
71        .validate_task(&task_id, &req.validator_id, &req.decision, req.reasoning.as_deref())
72        .await?;
73
74    let decision_str = match decision {
75        QuorumDecision::Verified => "verified",
76        QuorumDecision::Rejected => "rejected",
77        QuorumDecision::Disputed => "disputed",
78        QuorumDecision::Pending => "pending",
79    };
80
81    Ok(Json(ValidateResponse { decision: decision_str.to_string() }))
82}
83
84#[utoipa::path(
85    get,
86    path = "/v1/validations/{task_id}",
87    tag = "validation",
88    params(
89        ("task_id" = String, Path, description = "Task ID"),
90    ),
91    responses(
92        (status = 200, description = "Validation history", body = serde_json::Value),
93    ),
94)]
95pub async fn get_validations(
96    State(pool): State<SharedPool>,
97    Path(task_id): Path<String>,
98) -> Result<Json<serde_json::Value>, ac_types::error::AcError> {
99    let pool = pool.load();
100    let service = ValidationService::new(pool);
101    let history = service.get_validations(&task_id).await?;
102
103    Ok(Json(serde_json::json!({
104        "task_id": task_id,
105        "validations": history,
106        "protocol": "acp/1",
107    })))
108}
