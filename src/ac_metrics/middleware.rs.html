1//! Axum middleware that records HTTP request metrics.
2//!
3//! Provides the [`record_metrics`] middleware function which observes
4//! HTTP requests and records request count and duration to the
5//! Prometheus [`Metrics`] registry.
6
7use crate::registry::{MetricLabels, Metrics};
8use axum::{body::Body, extract::Extension, http::Request, middleware::Next, response::Response};
9use std::time::Instant;
10
11/// Axum middleware that records HTTP request metrics.
12///
13/// Records request count and duration with method, path, and status
14/// labels. Expects [`Metrics`] to be available via [`Extension`],
15/// set by the server during router construction.
16///
17/// # Example
18///
19/// ```ignore
20/// let app = Router::new()
21///     .layer(Extension(metrics))
22///     .layer(axum::middleware::from_fn(record_metrics));
23/// ```
24pub async fn record_metrics(
25    Extension(metrics): Extension<Metrics>,
26    req: Request<Body>,
27    next: Next,
28) -> Response {
29    let method = req.method().to_string();
30    let path = req.uri().path().to_string();
31    let start = Instant::now();
32
33    let resp = next.run(req).await;
34
35    let status = resp.status().as_u16().to_string();
36    let duration = start.elapsed().as_secs_f64();
37
38    let labels = MetricLabels { method, path, status };
39    metrics.http_requests_total.get_or_create(&labels).inc();
40    metrics.http_request_duration_seconds.get_or_create(&labels).observe(duration);
41
42    resp
43}
