1//! Prometheus registry and metric definitions.
2//!
3//! Defines the [`Metrics`] struct that holds a shared Prometheus registry
4//! and all pre-defined metrics for the Agent Commons server.
5
6use prometheus_client::{
7    encoding::{EncodeLabelKey, EncodeLabelSet, EncodeLabelValue, LabelSetEncoder},
8    metrics::{counter::Counter, family::Family, gauge::Gauge, histogram::Histogram},
9    registry::Registry,
10};
11use std::sync::Arc;

13/// Metric label key-value pairs for HTTP request metrics.
14///
15/// Used as the label type for `Family` metrics that track HTTP requests
16/// and request duration, broken down by method, path, and response status.
17#[derive(Clone, Debug, Hash, PartialEq, Eq)]
18pub struct MetricLabels {
19    /// HTTP method (GET, POST, etc.).
20    pub method: String,
21    /// Request path (e.g. `/v1/agents`).
22    pub path: String,
23    /// Response status code (e.g. `200`).
24    pub status: String,
25}
26
27impl EncodeLabelSet for MetricLabels {
28    fn encode(&self, mut encoder: LabelSetEncoder) -> Result<(), std::fmt::Error> {
29        // Encode method="...",path="...",status="..."
30        let mut label = encoder.encode_label();
31        let mut key_enc = label.encode_label_key()?;
32        EncodeLabelKey::encode(&"method", &mut key_enc)?;
33        let mut val_enc = key_enc.encode_label_value()?;
34        EncodeLabelValue::encode(&self.method, &mut val_enc)?;
35        val_enc.finish()?;
36
37        let mut label = encoder.encode_label();
38        let mut key_enc = label.encode_label_key()?;
39        EncodeLabelKey::encode(&"path", &mut key_enc)?;
40        let mut val_enc = key_enc.encode_label_value()?;
41        EncodeLabelValue::encode(&self.path, &mut val_enc)?;
42        val_enc.finish()?;
43
44        let mut label = encoder.encode_label();
45        let mut key_enc = label.encode_label_key()?;
46        EncodeLabelKey::encode(&"status", &mut key_enc)?;
47        let mut val_enc = key_enc.encode_label_value()?;
48        EncodeLabelValue::encode(&self.status, &mut val_enc)?;
49        val_enc.finish()?;
50
51        Ok(())
52    }
53}

55/// Label for task status metrics.
56///
57/// Wraps a single string value (e.g. `"running"`, `"completed"`, `"failed"`)
58/// used as the label for the `active_tasks` gauge family.
59#[derive(Clone, Debug, Hash, PartialEq, Eq)]
60pub struct TaskStatusLabel(pub String);
61
62impl EncodeLabelSet for TaskStatusLabel {
63    fn encode(&self, mut encoder: LabelSetEncoder) -> Result<(), std::fmt::Error> {
64        let mut label = encoder.encode_label();
65        let mut key_enc = label.encode_label_key()?;
66        EncodeLabelKey::encode(&"status", &mut key_enc)?;
67        let mut val_enc = key_enc.encode_label_value()?;
68        EncodeLabelValue::encode(&self.0, &mut val_enc)?;
69        val_enc.finish()?;
70        Ok(())
71    }
72}

74/// Histogram bucket boundaries in seconds.
75///
76/// Returns buckets from 1ms to 10s, suitable for measuring both fast
77/// database queries and slower HTTP request latencies.
78fn standard_buckets() -> impl Iterator<Item = f64> {
79    [0.001, 0.005, 0.01, 0.05, 0.1, 0.25, 0.5, 1.0, 2.5, 5.0, 10.0].into_iter()
80}

82/// Shared metrics holder.
83///
84/// Holds a Prometheus registry and all pre-defined metrics used by the
85/// Agent Commons server. Cloning is cheap (uses `Arc` internally for the
86/// registry; individual metrics use `Arc` within `Family` and `Histogram`).
87#[derive(Clone)]
88pub struct Metrics {
89    /// Prometheus registry shared via `Arc`.
90    pub registry: Arc<Registry>,
91    /// Total HTTP requests, labeled by [`MetricLabels`].
92    pub http_requests_total: Family<MetricLabels, Counter>,
93    /// HTTP request duration in seconds, labeled by [`MetricLabels`].
94    pub http_request_duration_seconds: Family<MetricLabels, Histogram>,
95    /// Currently active database pool connections.
96    pub db_pool_connections_active: Gauge,
97    /// Database query duration in seconds.
98    pub db_query_duration_seconds: Histogram,
99    /// Active tasks grouped by status, labeled by [`TaskStatusLabel`].
100    pub active_tasks: Family<TaskStatusLabel, Gauge>,
101    /// Unacknowledged messages pending in the mailbox.
102    pub messages_pending: Gauge,
103}
104
105impl Metrics {
106    /// Create a new [`Metrics`] with a fresh registry and all metrics
107    /// initialized to zero.
108    ///
109    /// Registers six metrics:
110    /// - `http_requests_total` — counter labeled by method, path, status
111    /// - `http_request_duration_seconds` — histogram labeled by method, path
112    /// - `db_pool_connections_active` — gauge for active DB connections
113    /// - `db_query_duration_seconds` — histogram for SQL query latency
114    /// - `active_tasks` — gauge family labeled by task status
115    /// - `messages_pending` — gauge for pending mailbox messages
116    #[allow(clippy::new_without_default)]
117    pub fn new() -> Self {
118        let mut registry = Registry::default();
119
120        let http_requests_total = Family::<MetricLabels, Counter>::default();
121        registry.register(
122            "http_requests_total",
123            "Total HTTP requests",
124            http_requests_total.clone(),
125        );
126
127        let http_request_duration_seconds =
128            Family::<MetricLabels, Histogram>::new_with_constructor(|| {
129                Histogram::new(standard_buckets())
130            });
131        registry.register(
132            "http_request_duration_seconds",
133            "HTTP request latency",
134            http_request_duration_seconds.clone(),
135        );
136
137        let db_pool_connections_active = Gauge::default();
138        registry.register(
139            "db_pool_connections_active",
140            "Active DB pool connections",
141            db_pool_connections_active.clone(),
142        );
143
144        let db_query_duration_seconds = Histogram::new(standard_buckets());
145        registry.register(
146            "db_query_duration_seconds",
147            "SQL query latency",
148            db_query_duration_seconds.clone(),
149        );
150
151        let active_tasks = Family::<TaskStatusLabel, Gauge>::default();
152        registry.register("active_tasks", "Tasks by status", active_tasks.clone());
153
154        let messages_pending = Gauge::default();
155        registry.register("messages_pending", "Unacknowledged messages", messages_pending.clone());
156
157        Self {
158            registry: Arc::new(registry),
159            http_requests_total,
160            http_request_duration_seconds,
161            db_pool_connections_active,
162            db_query_duration_seconds,
163            active_tasks,
164            messages_pending,
165        }
166    }
167
168    /// Encode all registered metrics to Prometheus text exposition format.
169    ///
170    /// Returns a `String` suitable for serving on the `/metrics` endpoint.
171    pub fn encode(&self) -> String {
172        let mut buffer = String::new();
173        prometheus_client::encoding::text::encode(&mut buffer, &self.registry)
174            .expect("failed to encode metrics");
175        buffer
176    }
177}
