3//! This is the `main` entry point for the `agent-commons` executable.
4//! It initializes tracing, loads runtime configuration, creates the

7//! # Environment Variables
8//!
9//! | Variable | Description | Default |
10//! |---|---|---|

12//! | `AC_HOST` | HTTP bind address | `0.0.0.0` |
13//! | `AC_PORT` | HTTP listen port | `3000` |
14//! | `AC_RDS_IAM_AUTH` | Enable RDS IAM auth token refresh | `false` |

18use ac_server::config::Settings;

23/// Binary entry point for the Agent Commons server.
24///
25/// Performs the following startup sequence:
26/// 1. Initialize the `tracing` subscriber for structured logging
27/// 2. Load runtime [`Settings`] from environment variables
28/// 3. Create a PostgreSQL connection pool (with optional RDS IAM auth)
29/// 4. Run database migrations
30/// 5. Build the axum [`Router`] with all feature sub-routes
31/// 6. Bind to the configured address and serve requests
32///
33/// When `AC_RDS_IAM_AUTH=true` and the `rds-iam` feature is enabled,
34/// the server generates an IAM auth token on startup and spawns a
35/// background task to refresh the token every ~12 minutes.
36///
37/// The server shuts down gracefully on SIGINT (Ctrl+C).
38#[tokio::main]
39async fn main() -> anyhow::Result<()> {
40    // Initialize structured logging
41    tracing_subscriber::fmt::init();
42
43    // Load configuration
44    let settings = match Settings::new() {
45        Ok(s) => s,
46        Err(e) => {
47            println!("Settings error: {:?}", e);
48            Settings::default()
49        }
50    };
51    println!("settings.rds_iam_auth = {}", settings.rds_iam_auth);
52
53    // Create database connection pool
54    println!("Creating shared pool...");
55    let shared_pool = create_pool(&settings).await?;
56    println!("Shared pool created successfully!");
57
58    // Run pending database migrations
59    let pool = shared_pool.load();
60    println!("Running migrations...");
61    sqlx::migrate!("../../migrations").run(&pool).await?;
62    println!("Migrations complete!");
63    drop(pool);
64
65    // Create metrics registry
66    let metrics = Metrics::new();
67
68    // Create rate limiter
69    let rate_limiter = RateLimiter::new(
70        settings.rate_limit_global,
71        settings.rate_limit_per_agent,
72        settings.rate_limit_window_secs,
73    );
74
75    // Build the application router
76    let app = create_app(shared_pool, metrics, rate_limiter);
77
78    tracing::info!("Agent Commons starting on {}:{}", settings.host, settings.port);
79
80    // Start the HTTP server
81    let listener =
82        tokio::net::TcpListener::bind(format!("{}:{}", settings.host, settings.port)).await?;
83    axum::serve(listener, app).with_graceful_shutdown(shutdown_signal()).await?;
84
85    Ok(())
86}

88/// Create the shared pool, optionally with RDS IAM auth and background
89/// token refresh.
90async fn create_pool(settings: &Settings) -> anyhow::Result<SharedPool> {
91    #[cfg(feature = "rds-iam")]
92    if settings.rds_iam_auth {
93        use ac_server::rds_iam;
94
95        let (pool, params) = rds_iam::create_iam_pool(&settings.database_url).await?;
96        let shared = SharedPool::new(pool);
97
98        // Spawn background token refresh loop
99        rds_iam::start_refresh_loop(shared.clone(), params, None);
100
101        tracing::info!("RDS IAM auth enabled — token refresh loop started");
102        return Ok(shared);
103    }
104
105    // Standard mode: static credentials, no refresh needed
106    let pool = PgPoolOptions::new().max_connections(10).connect(&settings.database_url).await?;
107    Ok(SharedPool::new(pool))
108}

110/// Wait for SIGINT (Ctrl+C) and return a future that resolves when received.
111async fn shutdown_signal() {
112    let ctrl_c = tokio::signal::ctrl_c();
113    ctrl_c.await.expect("failed to install Ctrl+C handler");
114    tracing::info!("shutdown signal received, draining connections...");
115}
