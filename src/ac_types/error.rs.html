1//! Application-level error types.
2//!
3//! [`AcError`] is the unified error enum used throughout the Agent Commons
4//! codebase.  Each variant carries a human-readable message suitable for
5//! logging and API responses.
6
7use thiserror::Error;
8
9/// Application-level errors for Agent Commons.
10///
11/// This enum covers the full range of expected failure modes: missing
12/// entities, cryptographic failures, constraint violations, database
13/// errors, and unexpected internal failures.
14#[derive(Debug, Error)]
15pub enum AcError {
16    /// The requested agent ID does not exist in the registry.
17    #[error("agent not found: {0}")]
18    AgentNotFound(
19        /// The agent ID that was looked up.
20        String,
21    ),
22
23    /// A cryptographic signature verification failed.
24    #[error("invalid signature: {0}")]
25    InvalidSignature(
26        /// Details about the verification failure.
27        String,
28    ),
29
30    /// An agent tried to register a public key that is already in use.
31    #[error("duplicate registration: {0}")]
32    DuplicateRegistration(
33        /// The public key that caused the conflict.
34        String,
35    ),
36
37    /// The requested task ID does not exist.
38    #[error("task not found: {0}")]
39    TaskNotFound(
40        /// The task ID that was looked up.
41        String,
42    ),
43
44    /// An attempt was made to transition a task to an invalid state.
45    #[error("invalid task state transition: {from:?} -> {to:?}")]
46    InvalidTaskTransition {
47        /// The current task status string.
48        from: String,
49        /// The requested target status string.
50        to: String,
51    },
52
53    /// Fewer validators responded than the task's required quorum.
54    #[error("insufficient validators: required {required}, got {got}")]
55    InsufficientValidators {
56        /// The number of validators the task required.
57        required: usize,
58        /// The number of validators that actually responded.
59        got: usize,
60    },
61
62    /// A wrapped database error from sqlx or the PostgreSQL driver.
63    #[error("database error: {0}")]
64    Database(
65        /// The underlying database error message.
66        String,
67    ),
68
69    /// An unexpected internal error occurred.
70    #[error("internal error: {0}")]
71    Internal(
72        /// Description of the internal failure.
73        String,
74    ),
75}
76
77impl axum::response::IntoResponse for AcError {
78    fn into_response(self) -> axum::response::Response {
79        let status = match &self {
80            AcError::AgentNotFound(_) | AcError::TaskNotFound(_) => {
81                axum::http::StatusCode::NOT_FOUND
82            }
83            AcError::DuplicateRegistration(_) => axum::http::StatusCode::CONFLICT,
84            AcError::InvalidTaskTransition { .. } => axum::http::StatusCode::BAD_REQUEST,
85            AcError::InvalidSignature(_) | AcError::InsufficientValidators { .. } => {
86                axum::http::StatusCode::BAD_REQUEST
87            }
88            _ => axum::http::StatusCode::INTERNAL_SERVER_ERROR,
89        };
90        (status, axum::Json(serde_json::json!({ "error": self.to_string() }))).into_response()
91    }
92}
