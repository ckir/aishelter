1//! Mailbox message types and envelope.
2//!
3//! Defines the [`MessageType`] enum for the various protocol-level messages
4//! agents exchange (task offers, accepts, results, etc.) and the [`Message`]
5//! struct that wraps them in a persistent, addressable envelope.
6
7use crate::agent::AgentId;
8use chrono::{DateTime, Utc};
9use serde::{Deserialize, Serialize};
10use uuid::Uuid;
11
12/// Message types for the agent protocol.
13///
14/// Each variant corresponds to a specific phase of the task lifecycle
15/// (§16) or a service-level announcement between agents.
16#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
17#[serde(rename_all = "snake_case")]
18pub enum MessageType {
19    /// Agent announced its presence on the network.
20    AgentHello,
21    /// Agent published or updated its capabilities.
22    AgentCapability,
23    /// A task is being offered to another agent.
24    TaskOffer,
25    /// The receiving agent accepts the offered task.
26    TaskAccept,
27    /// The receiving agent declines the offered task.
28    TaskReject,
29    /// The requester cancels a previously offered task.
30    TaskCancel,
31    /// Progress update from the executing agent.
32    TaskProgress,
33    /// The executing agent submits the task result.
34    TaskResult,
35    /// A validator submits a verification decision.
36    TaskVerify,
37    /// A validator raises a dispute about the result.
38    TaskDispute,
39    /// An agent offers a service it can perform.
40    ServiceOffer,
41    /// An agent requests a service from another agent.
42    ServiceRequest,
43}
44
45/// A message in the agent mailbox.
46///
47/// Messages are stored persistently in the database and delivered to the
48/// recipient agent when it next polls its mailbox (§15).  Each message has
49/// a unique UUID, sender, recipient, type, JSON payload, optional expiry,
50/// and an acknowledgement flag.
51#[derive(Debug, Clone, Serialize, Deserialize)]
52pub struct Message {
53    /// Unique message identifier (UUID v4).
54    pub message_id: Uuid,
55    /// The sending agent's ID.
56    pub from_agent_id: AgentId,
57    /// The receiving agent's ID.
58    pub to_agent_id: AgentId,
59    /// The kind of protocol message.
60    pub message_type: MessageType,
61    /// Arbitrary JSON payload specific to the message type.
62    pub payload: serde_json::Value,
63    /// When the message was created.
64    pub created_at: DateTime<Utc>,
65    /// Optional expiration — the message should be ignored after this time.
66    pub expires_at: Option<DateTime<Utc>>,
67    /// Whether the recipient has acknowledged reading this message.
68    pub acknowledged: bool,
69    /// When the message was acknowledged (if it has been).
70    pub acknowledged_at: Option<DateTime<Utc>>,
71}
