1//! SQL migration runner.
2//!
3//! Provides [`run_migrations`] to apply all pending SQL migrations from the
4//! `migrations/` directory to the connected PostgreSQL database.  Uses
5//! `sqlx::migrate!` which embeds the migration files at compile time and
6//! tracks applied migrations in the `sqlx_migrations` table.
7
8use sqlx::PgPool;
9
10/// Run SQL migrations from the embedded migrations directory.
11///
12/// Applies any migrations from `migrations/` that have not yet been
13/// recorded in the database's `sqlx_migrations` tracking table.  Migrations
14/// are applied in filename order within a single transaction per migration
15/// file (§8 — Initial schema in `001_initial_schema.sql`).
16///
17/// # Arguments
18///
19/// * `pool` — Active PostgreSQL connection pool.
20///
21/// # Errors
22///
23/// Returns an error if any migration SQL is invalid, if the migration
24/// directory is missing, or if a migration fails mid-transaction.
25pub async fn run_migrations(
26    // Active PostgreSQL connection pool.
27    pool: &PgPool,
28) -> anyhow::Result<()> {
29    // Log that migrations are starting for observability.
30    tracing::info!("Running database migrations");
31    // Run embedded migrations from the ../../migrations directory.
32    // The path is relative to this source file at compile time.
33    sqlx::migrate!("../../migrations").run(pool).await?;
34    Ok(())
35}
